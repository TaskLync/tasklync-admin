// middleware.ts
import { NextRequest, NextResponse } from 'next/server'
import { verifyAccessToken } from '@/lib/auth/jwt'
import { permissionForPath, hasPermission } from '@/lib/auth/rbac'
import { COOKIE } from '@/lib/auth/cookies'
import { safeNextPath } from '@/lib/auth/redirect'
import { hasInvalidIdSegment } from '@/lib/auth/uuid-guard'
import type { AccessTokenPayload } from '@/types/auth'

const UNSAFE_METHODS = new Set(['POST', 'PUT', 'PATCH', 'DELETE'])

function json(status: number, body: Record<string, unknown>) {
  return NextResponse.json(body, { status })
}

// CSRF defence in depth (cookies are SameSite=Lax): a browser always sends Origin on
// cross-site unsafe requests, and it must match our own host.
function isSameOrigin(request: NextRequest): boolean {
  const origin = request.headers.get('origin')
  if (!origin) return true // non-browser clients; they would still need the cookies
  try {
    const host = request.headers.get('x-forwarded-host') ?? request.headers.get('host')
    return new URL(origin).host === host
  } catch {
    return false
  }
}

export async function middleware(request: NextRequest) {
  const { pathname, search } = request.nextUrl
  const isApi = pathname.startsWith('/api/')
  const redirect = (to: string) => NextResponse.redirect(new URL(to, request.url))

  if (isApi && UNSAFE_METHODS.has(request.method) && !isSameOrigin(request)) {
    return json(403, { error: 'Forbidden' })
  }

  // Auth endpoints enforce their own rules.
  if (pathname.startsWith('/api/admin/auth')) return NextResponse.next()

  const accessToken = request.cookies.get(COOKIE.access)?.value
  const hasRefresh = Boolean(request.cookies.get(COOKIE.session)?.value)

  let payload: AccessTokenPayload | null = null
  if (accessToken) {
    try {
      payload = await verifyAccessToken(accessToken) // sig, alg, iss, aud, exp, type, scope
    } catch {
      payload = null
    }
  }

  // ---- /auth/* pages --------------------------------------------------------
  if (pathname.startsWith('/auth')) {
    if (payload?.scope === 'full') return redirect('/')
    if (payload?.scope === 'mfa_setup' && pathname !== '/auth/mfa-setup') return redirect('/auth/mfa-setup')
    if (!payload && hasRefresh && pathname === '/auth/login' && !request.nextUrl.searchParams.has('reason')) {
      const next = safeNextPath(request.nextUrl.searchParams.get('next'))
      return redirect(`/api/admin/auth/refresh?next=${encodeURIComponent(next)}`)
    }
    return NextResponse.next()
  }

  // ---- protected pages + APIs ----------------------------------------------
  if (!payload) {
    if (isApi) return json(401, { error: 'Unauthorized' })
    if (hasRefresh) {
      return redirect(`/api/admin/auth/refresh?next=${encodeURIComponent(pathname + search)}`)
    }
    const url = new URL('/auth/login', request.url)
    if (pathname !== '/') url.searchParams.set('next', pathname + search)
    return NextResponse.redirect(url)
  }

  if (payload.scope !== 'full') {
    return isApi
      ? json(403, { error: 'MFA setup required', code: 'MFA_SETUP_REQUIRED' })
      : redirect('/auth/mfa-setup')
  }

  // A malformed id is a 404, not a SQL error (audit M-22).
  if (isApi && hasInvalidIdSegment(pathname)) return json(404, { error: 'Not found' })

  const requiredPermission = permissionForPath(pathname)
  if (requiredPermission && !hasPermission(payload.role, requiredPermission)) {
    return isApi ? json(403, { error: 'Forbidden' }) : redirect('/')
  }

  return NextResponse.next()
}

CREATE TABLE IF NOT EXISTS public.booking_status_history (

  from_status   varchar(30) NULL,

CREATE OR REPLACE FUNCTION public.fn_log_booking_status_history() RETURNS trigger AS $$
DECLARE
  v_actor text;
BEGIN
  -- Admin transactions run: SELECT set_config('app.actor', 'admin:<email>', true)
  -- Everything else (booking-service) leaves it unset -> NULL.
  v_actor := NULLIF(current_setting('app.actor', true), '');

  IF TG_OP = 'INSERT' THEN
    INSERT INTO booking_status_history (booking_id, from_status, to_status, actor, changed_at)
    VALUES (NEW.id, NULL, NEW.status::text, v_actor, NEW.created_at);
  ELSIF NEW.status IS DISTINCT FROM OLD.status THEN
    INSERT INTO booking_status_history (booking_id, from_status, to_status, actor, changed_at)
    VALUES (NEW.id, OLD.status::text, NEW.status::text, v_actor, now());
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_booking_status_history ON public.bookings;
CREATE TRIGGER trg_booking_status_history
  AFTER INSERT OR UPDATE OF status ON public.bookings
  FOR EACH ROW EXECUTE FUNCTION public.fn_log_booking_status_history();

-- Backfill: bookings that existed before this trigger get an approximate history
-- (created as PENDING, then one jump to the current status at updated_at).
INSERT INTO public.booking_status_history (booking_id, from_status, to_status, is_backfilled, changed_at)
SELECT b.id, NULL, 'PENDING', true, b.created_at
FROM public.bookings b
WHERE NOT EXISTS (SELECT 1 FROM public.booking_status_history h WHERE h.booking_id = b.id);

INSERT INTO public.booking_status_history (booking_id, from_status, to_status, is_backfilled, changed_at)
SELECT b.id, 'PENDING', b.status::text, true, b.updated_at
FROM public.bookings b
WHERE b.status::text <> 'PENDING'
  AND (SELECT count(*) FROM public.booking_status_history h WHERE h.booking_id = b.id) = 1;

-- ---------------------------------------------------------
-- 4. KPI cards for the list page. "Today" uses Asia/Karachi (same as workers).
-- ---------------------------------------------------------
DROP FUNCTION IF EXISTS public.get_booking_kpis();
CREATE OR REPLACE FUNCTION public.get_booking_kpis()
RETURNS TABLE (
  bookings_today       bigint,
  bookings_yesterday   bigint,
  pending_now          bigint,
  in_progress_now      bigint,
  disputed_now         bigint,
  created_sparkline_7d bigint[]
) AS $$
  WITH b AS (
    SELECT (date_trunc('day', now() AT TIME ZONE 'Asia/Karachi') AT TIME ZONE 'Asia/Karachi') AS today_start
  )
  SELECT
    (SELECT count(*) FROM bookings x WHERE x.created_at >= b.today_start),
    (SELECT count(*) FROM bookings x
       WHERE x.created_at >= b.today_start - interval '1 day'
         AND x.created_at <  b.today_start),
    (SELECT count(*) FROM bookings x WHERE x.status::text = 'PENDING'),
    (SELECT count(*) FROM bookings x WHERE x.status::text = 'IN_PROGRESS'),
    (SELECT count(*) FROM bookings x WHERE x.status::text = 'DISPUTED'),
    (SELECT array_agg(s.c ORDER BY s.d)
       FROM (
         SELECT g.d,
                (SELECT count(*) FROM bookings x
                   WHERE x.created_at >= b.today_start - (6 - g.d) * interval '1 day'
                     AND x.created_at <  b.today_start - (5 - g.d) * interval '1 day') AS c
         FROM generate_series(0, 6) AS g(d)
       ) s)
  FROM b;
$$ LANGUAGE sql STABLE;

-- 4. One date basis for money everywhere: completion time, Asia/Karachi.
--    (the dashboard used payments.created_at / UTC dates, finance used completion time)
-- ───────────────────────────────────────────────────────────
CREATE OR REPLACE FUNCTION get_dashboard_kpis()
RETURNS JSON AS $$
DECLARE
  v_today  date := (now() AT TIME ZONE 'Asia/Karachi')::date;
  v_result JSON;
BEGIN
  SELECT json_build_object(
    'revenueToday', (
      SELECT COALESCE(SUM(b.estimated_total), 0) FROM bookings b
       WHERE b.status IN ('COMPLETED', 'AUTO_COMPLETED')
         AND (COALESCE(b.completed_at, b.updated_at) AT TIME ZONE 'Asia/Karachi')::date = v_today
    ),
    'revenueYesterday', (
      SELECT COALESCE(SUM(b.estimated_total), 0) FROM bookings b
       WHERE b.status IN ('COMPLETED', 'AUTO_COMPLETED')
         AND (COALESCE(b.completed_at, b.updated_at) AT TIME ZONE 'Asia/Karachi')::date = v_today - 1
    ),
    'revenueSparkline', (
      SELECT COALESCE(json_agg(day_total ORDER BY day), '[]')
      FROM (
        SELECT d::date AS day,

CREATE OR REPLACE FUNCTION get_revenue_chart(p_period TEXT DEFAULT 'daily')
RETURNS TABLE (bucket TIMESTAMPTZ, gross_revenue NUMERIC, platform_fees NUMERIC) AS $$
DECLARE
  v_unit TEXT;
  v_from TIMESTAMPTZ;
BEGIN
  IF p_period = 'weekly' THEN
    v_unit := 'week';  v_from := now() - INTERVAL '12 weeks';
  ELSIF p_period = 'monthly' THEN
    v_unit := 'month'; v_from := now() - INTERVAL '12 months';
  ELSE
    v_unit := 'day';   v_from := now() - INTERVAL '30 days';
  END IF;

  RETURN QUERY
    SELECT (date_trunc(v_unit, COALESCE(b.completed_at, b.updated_at) AT TIME ZONE 'Asia/Karachi')
              AT TIME ZONE 'Asia/Karachi'),
           COALESCE(SUM(b.estimated_total), 0),
           COALESCE(SUM(b.platform_fee), 0)
      FROM bookings b
     WHERE b.status IN ('COMPLETED', 'AUTO_COMPLETED')
       AND COALESCE(b.completed_at, b.updated_at) > v_from
     GROUP BY 1
     ORDER BY 1;
END;
$$ LANGUAGE plpgsql STABLE;

const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i

// entity prefix -> static sub-routes that are NOT an id
const GUARDS: { prefix: string; statics: string[] }[] = [
  { prefix: '/api/admin/users/', statics: ['kpis', 'export'] },
  { prefix: '/api/admin/workers/', statics: ['kpis', 'export', 'filter-options', 'verification'] },
  { prefix: '/api/admin/bookings/', statics: ['kpis', 'export', 'board'] },
  { prefix: '/api/admin/disputes/', statics: ['counts', 'refunds'] },
  { prefix: '/api/admin/reviews/', statics: ['counts', 'audit'] },
  { prefix: '/api/admin/payments/payouts/', statics: [] },
  { prefix: '/api/admin/notifications/campaigns/', statics: [] },
  { prefix: '/api/admin/admins/', statics: [] },
]

/** true when the path has an id segment that is not a UUID. */
export function hasInvalidIdSegment(pathname: string): boolean {
  for (const g of GUARDS) {
    if (!pathname.startsWith(g.prefix)) continue
    const segment = pathname.slice(g.prefix.length).split('/')[0]
    if (!segment || g.statics.includes(segment)) return false
    return !UUID_RE.test(segment)
  }
  return false
}

// lib/rbac-guard.ts
import { NextResponse } from 'next/server'
import { cookies } from 'next/headers'
import { authenticateAccessToken } from '@/lib/auth/request'
import { hasPermission } from '@/lib/auth/rbac'
import { COOKIE } from '@/lib/auth/cookies'
import type { AdminRole } from '@/types/auth'

interface AdminContext {
  adminId: string
  adminEmail: string
  role: AdminRole
  sessionId: string
}

type PermissionResult =
  | { ok: true; admin: AdminContext }
  | { ok: false; response: NextResponse }

export async function requirePermission(permission: string): Promise<PermissionResult> {
  const cookieStore = await cookies()
  const payload = await authenticateAccessToken(cookieStore.get(COOKIE.access)?.value)

  if (!payload) {
    return { ok: false, response: NextResponse.json({ error: 'Unauthorized' }, { status: 401 }) }
  }

  // A password-only session (MFA not enrolled yet) can never reach admin data.
  if (payload.scope !== 'full') {
    return {
      ok: false,
      response: NextResponse.json({ error: 'MFA setup required', code: 'MFA_SETUP_REQUIRED' }, { status: 403 }),
    }
  }

  if (!hasPermission(payload.role, permission)) {
    return { ok: false, response: NextResponse.json({ error: 'Forbidden' }, { status: 403 }) }
  }

  return {
    ok: true,
    admin: {
      adminId: payload.sub,
      adminEmail: payload.email,
      role: payload.role,
      sessionId: payload.sessionId,
    },
  }
}

// lib/internal-client.ts  (server only)
import { NextResponse } from 'next/server'

export class ServiceError extends Error {
  status: number
  code?: string
  service?: string
  constructor(status: number, message: string, code?: string, service?: string) {
    super(message)
    this.status = status
    this.code = code
    this.service = service
  }
}

export interface ServiceTarget {
  name: string
  url?: string
  key?: string
}

interface CallOptions {
  body?: object
  query?: Record<string, string | number | undefined>
  timeoutMs?: number
}

export async function callService<T = unknown>(
  target: ServiceTarget,
  method: 'GET' | 'POST' | 'PATCH',
  path: string,
  opts: CallOptions = {}
): Promise<T> {
  if (!target.url || !target.key) {
    throw new ServiceError(503, `${target.name} is not configured on the admin server.`, 'NOT_CONFIGURED', target.name)
  }

  const url = new URL(target.url.replace(/\/$/, '') + path)
  for (const [k, v] of Object.entries(opts.query ?? {})) {
    if (v !== undefined) url.searchParams.set(k, String(v))
  }

  const ctrl = new AbortController()
  const timer = setTimeout(() => ctrl.abort(), opts.timeoutMs ?? 15_000)

  let res: Response
  try {
    res = await fetch(url, {
      method,
      headers: { 'Content-Type': 'application/json', 'x-internal-key': target.key },
      body: opts.body ? JSON.stringify(opts.body) : undefined,
      signal: ctrl.signal,
      cache: 'no-store',
    })
  } catch {
    throw new ServiceError(503, `${target.name} is unreachable. Try again in a moment.`, 'UNREACHABLE', target.name)
  } finally {
    clearTimeout(timer)
  }

  const text = await res.text()
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  let json: any = null
  try {
    json = text ? JSON.parse(text) : null
  } catch {
    /* not JSON */
  }

  if (!res.ok) {
    const message =
      json?.error?.message ??
      (typeof json?.error === 'string' ? json.error : null) ??
      json?.message ??
      `${target.name} answered HTTP ${res.status}`
    throw new ServiceError(res.status, message, json?.error?.code, target.name)
  }
  return (json?.data ?? json) as T
}

/** Maps any error from an action route to a JSON response. 5xx from a service becomes 502. */
export function serviceErrorResponse(err: unknown, tag: string) {
  if (err instanceof ServiceError) {
    const status = err.status >= 500 ? 502 : err.status
    if (err.status >= 500) console.error(tag, err.service, err.status, err.message)
    return NextResponse.json({ error: err.message, code: err.code }, { status })
  }
  const code = (err as { code?: string } | null)?.code
  if (code === '23505') {
    return NextResponse.json({ error: 'A conflicting record already exists.' }, { status: 409 })
  }
  console.error(tag, err)
  return NextResponse.json({ error: 'Action failed' }, { status: 500 })
}

/** Used for best-effort follow-ups (search reindex, cache clear). Never throws. */
export async function bestEffort(label: string, fn: () => Promise<unknown>): Promise<boolean> {
  try {
    await fn()
    return true
  } catch (err) {
    console.warn(`[best-effort] ${label} failed:`, err instanceof Error ? err.message : err)
    return false
  }
}

import { Pool } from 'pg'

  var _pgPool: Pool | undefined

function sslConfig() {
  const url = process.env.DATABASE_URL ?? ''
  const wantsSsl =
    url.includes('supabase') ||
    url.includes('aws') ||
    process.env.DATABASE_SSL === 'true' ||
    process.env.NODE_ENV === 'production'
  if (!wantsSsl) return undefined

  const ca = process.env.DATABASE_CA_CERT
  if (ca) {
    const pem = ca.includes('BEGIN CERTIFICATE')
      ? ca.replace(/\\n/g, '\n')
      : Buffer.from(ca, 'base64').toString('utf8')
    return { ca: pem, rejectUnauthorized: true }
  }

  if (process.env.NODE_ENV === 'production' && process.env.NEXT_PHASE !== 'phase-production-build') {
    console.warn('[db] DATABASE_CA_CERT is not set: TLS is encrypted but the server certificate is NOT verified.')
  }
  return { rejectUnauthorized: false }
}

export const db =
  global._pgPool ??
  new Pool({
    connectionString: process.env.DATABASE_URL,
    max: 10,
    idleTimeoutMillis: 30000,
    ssl: sslConfig(),
  })

if (process.env.NODE_ENV !== 'production') global._pgPool = db

// lib/auth-fetch.ts  (client only)
// Silent token refresh for EVERY same-origin /api/admin/* call, so none of your
// existing hooks/mutations have to change. A 401 -> refresh once -> retry once.

const LAST_KEY = 'admin_last_refresh'
const EXCLUDED = [
  '/api/admin/auth/login',
  '/api/admin/auth/refresh',
  '/api/admin/auth/logout',
  '/api/admin/auth/mfa/verify',
]

let inFlight: Promise<boolean> | null = null

function lastRefreshAt(): number {
  try {
    return Number(localStorage.getItem(LAST_KEY) ?? 0)
  } catch {
    return 0
  }
}

function markRefreshed() {
  try {
    localStorage.setItem(LAST_KEY, String(Date.now()))
  } catch {
    /* private mode etc. */
  }
}

async function callRefresh(force: boolean): Promise<boolean> {
  // Another tab (or this one) refreshed a moment ago: cookies are shared, just retry.
  if (!force && Date.now() - lastRefreshAt() < 8000) return true
  const res = await window.fetch('/api/admin/auth/refresh', { method: 'POST', credentials: 'same-origin' })
  if (res.ok) markRefreshed()
  return res.ok
}

export function refreshSession(force = false): Promise<boolean> {
  if (inFlight) return inFlight
  const run = (): Promise<boolean> => callRefresh(force)
  // Web Locks serialise refreshes across tabs; fall back gracefully where unsupported.
  // The async wrapper + await flattens the lock's nested promise type to a plain boolean.
  const p: Promise<boolean> = (async (): Promise<boolean> => {
    if (typeof navigator !== 'undefined' && 'locks' in navigator) {
      return await navigator.locks.request('admin-auth-refresh', run)
    }
    return run()
  })()
  const tracked: Promise<boolean> = p.catch(() => false).finally(() => {
    inFlight = null
  })
  inFlight = tracked
  return tracked
}

export function getLastRefreshAt() {
  return lastRefreshAt()
}

function redirectToLogin() {
  if (window.location.pathname.startsWith('/auth/login')) return
  window.location.assign('/auth/login?reason=expired')
}

function shouldIntercept(url: string): boolean {
  try {
    const u = new URL(url, window.location.origin)
    if (u.origin !== window.location.origin) return false
    if (!u.pathname.startsWith('/api/admin/')) return false
    return !EXCLUDED.includes(u.pathname)
  } catch {
    return false
  }
}

const FLAG = '__tasklyncAuthFetch'

export function installFetchInterceptor() {
  if (typeof window === 'undefined') return
  const w = window as unknown as Record<string, unknown>
  if (w[FLAG]) return
  w[FLAG] = true

  const originalFetch = window.fetch.bind(window)

  window.fetch = (async (input: RequestInfo | URL, init?: RequestInit) => {
    const url = typeof input === 'string' ? input : input instanceof URL ? input.href : input.url
    if (!shouldIntercept(url)) return originalFetch(input, init)

    const retryInput = input instanceof Request ? input.clone() : input
    const res = await originalFetch(input, init)
    if (res.status !== 401) return res

    const ok = await refreshSession()
    if (!ok) {
      redirectToLogin()
      return res
    }
    return originalFetch(retryInput, init)
  }) as typeof window.fetch
}

  }
  const code = (err as { code?: string } | null)?.code

      { status: 409 }
    )
  }

  return NextResponse.json({ error: 'Action failed' }, { status: 500 })
}

export function buildBookingWhere(
  sp: URLSearchParams,
  opts: { ignoreStatus?: boolean } = {}
): { where: string[]; params: unknown[] } {
  const where: string[] = []
  const params: unknown[] = []
  const push = (value: unknown) => {
    params.push(value)
    return `$${params.length}`
  }

  const search = (sp.get('search') || '').trim()
  const status = sp.get('status') || 'all'
  const city = sp.get('city') || 'all'
  const categoryId = sp.get('categoryId') || 'all'
  const paymentStatus = sp.get('paymentStatus') || 'all'
  const hasDispute = sp.get('hasDispute') || 'all'
  const amountMin = Number(sp.get('amountMin'))
  const amountMax = Number(sp.get('amountMax'))
  const dateFrom = sp.get('dateFrom')
  const dateTo = sp.get('dateTo')

  if (search) {
    const p = push(`%${search}%`)
    where.push(
      `(b.id::text ILIKE ${p} OR u.name ILIKE ${p} OR u.phone ILIKE ${p} OR w.name ILIKE ${p} OR w.phone ILIKE ${p})`
    )
  }

  if (!opts.ignoreStatus && (BOOKING_STATUSES as readonly string[]).includes(status)) {
    where.push(`b.status::text = ${push(status)}`)
  }

  if (city !== 'all') {
    where.push(`${CITY_SQL} = ${push(city)}`)
  }

  if (categoryId !== 'all' && CATEGORY_RE.test(categoryId)) {
    where.push(`b.category_id = ${push(categoryId)}`)
  }

  if ((PAYMENT_STATUSES as readonly string[]).includes(paymentStatus)) {
    where.push(`${PAYMENT_STATUS_SQL} = ${push(paymentStatus)}`)
  }

  if (hasDispute === 'yes') {
    where.push('EXISTS (SELECT 1 FROM disputes d WHERE d.booking_id = b.id)')
  } else if (hasDispute === 'no') {
    where.push('NOT EXISTS (SELECT 1 FROM disputes d WHERE d.booking_id = b.id)')
  }

  if (sp.get('amountMin') && Number.isFinite(amountMin)) {
    where.push(`b.estimated_total >= ${push(amountMin)}`)
  }
  if (sp.get('amountMax') && Number.isFinite(amountMax)) {
    where.push(`b.estimated_total <= ${push(amountMax)}`)
  }

  if (dateFrom && DATE_RE.test(dateFrom)) {
    where.push(`b.created_at >= ${push(dateFrom)}::date`)
  }
  if (dateTo && DATE_RE.test(dateTo)) {
    // inclusive of the whole "to" day
    where.push(`b.created_at < (${push(dateTo)}::date + 1)`)
  }

  return { where, params }
}

export const toWhereSql = (where: string[]) => (where.length ? `WHERE ${where.join(' AND ')}` : '')

 */
export function standardRefundPercent(
  status: BookingStatus,
  scheduledAt: string | Date,
  now: Date = new Date()

  if (status === 'PENDING') return 100

// lib/notification-service.ts
// Server-only. Calls the API gateway, which forwards to notification-service.
import type { BroadcastCategory, DeliveryChannel, SegmentType } from '@/lib/notification-constants'

export class NotificationServiceError extends Error {
  status: number
  code: string
  constructor(status: number, code: string, message: string) {
    super(message)
    this.status = status
    this.code = code
  }
}

export interface ServiceBroadcastPayload {
  title: string
  template_key: string
  template_data: Record<string, unknown>
  channels: DeliveryChannel[]
  category: BroadcastCategory
  segment_type: SegmentType
  segment_filter: Record<string, unknown>
  created_by: string
}

export async function launchBroadcast(
  payload: ServiceBroadcastPayload
): Promise<{ campaign_id: string; status: string }> {
  const base = process.env.API_GATEWAY_URL
  const adminKey = process.env.GATEWAY_ADMIN_KEY
  if (!base || !adminKey) {
    throw new NotificationServiceError(
      503,
      'NOT_CONFIGURED',
      'API_GATEWAY_URL or GATEWAY_ADMIN_KEY is not set on the admin panel.'
    )
  }

  const url = `${base.replace(/\/$/, '')}/internal-admin/notifications/broadcasts`
  const ctrl = new AbortController()
  const timer = setTimeout(() => ctrl.abort(), 10_000)

  let res: Response
  try {
    res = await fetch(url, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-admin-internal-key': adminKey,
      },
      body: JSON.stringify(payload),
      signal: ctrl.signal,
      cache: 'no-store',
    })
  } catch (err) {
    console.error('[notification-service] request failed', url, err)
    throw new NotificationServiceError(
      503,
      'UNREACHABLE',
      'API gateway is unreachable. Check that it is running and reachable from the admin server.'
    )
  } finally {
    clearTimeout(timer)
  }

  const text = await res.text()
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  let json: any = null
  try {
    json = JSON.parse(text)
  } catch {
    /* response was not JSON */
  }

  if (!res.ok) {
    console.error('[notification-service] broadcast rejected', res.status, text.slice(0, 1000))
    const detail =
      json?.error?.message ??
      (typeof json?.error === 'string' ? json.error : null) ??
      json?.message ??
      `HTTP ${res.status}: ${text.slice(0, 200)}`
    throw new NotificationServiceError(res.status, json?.error?.code ?? 'SERVICE_ERROR', detail)
  }

  // Accept { success, data: { campaign_id } } or a bare { campaign_id }.
  const data = json?.data ?? json
  if (!data?.campaign_id) {
    console.error('[notification-service] unexpected response', res.status, text.slice(0, 1000))
    throw new NotificationServiceError(
      502,
      'BAD_RESPONSE',
      `Unexpected response from service: ${text.slice(0, 200)}`
    )
  }
  return data
}

// ---------- Helpers ----------

export const getTemplate = (id: string) => CATALOG.find((t) => t.id === id)
export const templateKey = (id: string, lang: TemplateLanguage) => `${id}_${lang}`

export function parseTemplateKey(key: string): { template: CatalogTemplate; language: TemplateLanguage } | null {
  const m = /^(.+)_(en|ur)$/.exec(key)
  const template = m ? getTemplate(m[1]) : undefined
  return template && m ? { template, language: m[2] as TemplateLanguage } : null
}

const VAR_RE = /\{\{\s*(\w+)\s*\}\}/g

export function renderCopy(
  text: string,
  values: Record<string, string | number | undefined | null>,
  fallback?: (key: string) => string
): string {
  return text.replace(VAR_RE, (_m, key: string) => {
    const v = values[key]
    if (v !== undefined && v !== null && v !== '') return String(v)
    return fallback ? fallback(key) : `{{${key}}}`
  })
}

export function defaultValues(t: CatalogTemplate): Record<string, string> {
  const out: Record<string, string> = {}
  for (const v of t.variables) out[v.key] = v.type === 'year' ? String(new Date().getFullYear()) : ''
  return out
}

/** Validates + coerces variable values. Used by the launch route and the composer. */
export function checkValues(
  t: CatalogTemplate,
  raw: Record<string, unknown>
): { data: Record<string, string | number>; errors: string[] } {
  const data: Record<string, string | number> = {}
  const errors: string[] = []

  for (const v of t.variables) {
    const s = String(raw[v.key] ?? '').trim()
    if (!s) {
      if (v.required !== false) errors.push(`${v.label} is required`)
      continue
    }
    if (v.type === 'text') {
      if (s.length > 40) errors.push(`${v.label} must be 40 characters or fewer`)
      else if (/["\\]/.test(s)) errors.push(`${v.label} can't contain double quotes or backslashes`)
      else data[v.key] = s
    } else {
      const n = Number(s)
      const bad =
        !Number.isFinite(n) || (v.min !== undefined && n < v.min) || (v.max !== undefined && n > v.max)
      if (bad) errors.push(`${v.label} must be a number between ${v.min ?? '−∞'} and ${v.max ?? '∞'}`)
      else data[v.key] = n
    }
  }
  return { data, errors }
}

export type PreviewMessage = { title: string; body: string }

/** What the push will look like. Missing variables show as [label] so the admin sees what's left to fill. */
export function previewTemplate(
  t: CatalogTemplate,
  lang: TemplateLanguage,
  values: Record<string, string | number | undefined>
): PreviewMessage {
  const label = (key: string) => `[${t.variables.find((v) => v.key === key)?.label ?? key}]`
  const c = t.copy[lang]
  return { title: renderCopy(c.title, values, label), body: renderCopy(c.body, values, label) }
}

/** Rebuilds the message a past campaign sent, from its template_key + template_data. */
export function resolveCampaignMessage(
  key: string,
  data: Record<string, unknown>
): { title: string; body: string; label: string } | null {
  if (key === CUSTOM_TEMPLATE_KEY) {
    return { title: String(data.title ?? ''), body: String(data.body ?? ''), label: 'Custom message' }
  }
  const parsed = parseTemplateKey(key)
  if (!parsed) return null
  const values = data as Record<string, string | number>
  const p = previewTemplate(parsed.template, parsed.language, values)
  return { ...p, label: `${parsed.template.name} · ${LANGUAGE_LABEL[parsed.language]}` }
}

export const LIMITS = {
  label: 200, // broadcast_campaigns.title is VARCHAR(200)
  title: 65,
  pushBody: 178,
  smsBody: 150, // SMS template prefixes "TaskLync: "
  specificPeople: 500,
} as const

/** notification-service dedups (recipient, template_key) for 30s, so refuse a relaunch inside 35s. */
export const DUPLICATE_LAUNCH_WINDOW_SECONDS = 35

/** Matches QUIET_HOURS_* in notification-service. Normal-priority push/SMS is DROPPED in this window. */
export const QUIET_HOURS = { startHour: 22, endHour: 8, timezone: 'Asia/Karachi' } as const

export function karachiHour(date: Date = new Date()): number {
  const part = new Intl.DateTimeFormat('en-GB', {
    timeZone: QUIET_HOURS.timezone,
    hour: '2-digit',
    hourCycle: 'h23',
  })
    .formatToParts(date)
    .find((p) => p.type === 'hour')
  return Number(part?.value ?? 0) % 24
}

export function isQuietHours(date: Date = new Date()): boolean {
  const h = karachiHour(date)
  return h >= QUIET_HOURS.startHour || h < QUIET_HOURS.endHour
}

// ---------- Errors ----------

export class NotificationActionError extends Error {
  status: number
  constructor(status: number, message: string) {
    super(message)
    this.status = status
  }
}

export function notificationsErrorResponse(err: unknown, tag: string) {
  if (err instanceof NotificationActionError) {
    return NextResponse.json({ error: err.message }, { status: err.status })
  }
  if (err instanceof NotificationServiceError) {
    const status = err.status >= 500 ? 502 : err.status
    return NextResponse.json({ error: err.message, code: err.code }, { status })
  }
  const code = (err as { code?: string } | null)?.code
  if (code === '42P01' || code === '42703') {
    console.error(tag, err)
    return NextResponse.json(
      {
        error:
          'Notification tables are missing or out of date. Run notification-service migrations 003, 005 and 006 on the shared database.',
      },
      { status: 503 }
    )
  }
  const message = err instanceof Error ? err.message : String(err || 'Request failed')
  console.error(tag, err)
  return NextResponse.json({ error: message }, { status: 500 })
}

// ---------- Parsing + validation (never trust the browser) ----------

export function parseAudience(raw: unknown): AudienceSelection {
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  const a = raw as any
  const type = a?.segmentType

  if (type === 'all_users' || type === 'all_workers') return { segmentType: type }

  if (type === 'category_city') {
    const categoryId = String(a.categoryId ?? '').trim()
    const city = String(a.city ?? '').trim()
    if (!CATEGORY_RE.test(categoryId)) throw new NotificationActionError(400, 'Choose a service category')
    if (!city || city.length > 60) throw new NotificationActionError(400, 'Choose a city')
    return { segmentType: 'category_city', categoryId, city }
  }

  if (type === 'custom_ids') {
    const audience = a.audience === 'workers' ? 'workers' : a.audience === 'users' ? 'users' : null
    if (!audience) throw new NotificationActionError(400, 'Choose users or workers')
    const ids: string[] = Array.isArray(a.ids) ? [...new Set<string>(a.ids.map(String))] : []
    if (ids.length === 0) throw new NotificationActionError(400, 'Add at least one recipient')
    if (ids.length > LIMITS.specificPeople) {
      throw new NotificationActionError(400, `You can target at most ${LIMITS.specificPeople} specific people`)
    }
    if (!ids.every(isUuid)) throw new NotificationActionError(400, 'Recipient list contains an invalid ID')
    return { segmentType: 'custom_ids', audience, ids }
  }

  throw new NotificationActionError(400, 'Choose who should receive this')
}

export function parseCategory(raw: unknown): BroadcastCategory {
  if ((BROADCAST_CATEGORIES as readonly string[]).includes(String(raw))) return raw as BroadcastCategory
  throw new NotificationActionError(400, 'Choose a category: operational announcement or marketing')
}

export function parseChannels(raw: unknown): DeliveryChannel[] {
  const list = Array.isArray(raw) ? [...new Set(raw.map(String))] : []
  if (list.length === 0 || !list.every((c) => (DELIVERY_CHANNELS as readonly string[]).includes(c))) {
    throw new NotificationActionError(400, 'Choose at least one channel (push or SMS)')
  }
  return list as DeliveryChannel[]
}

function cleanText(value: unknown, label: string, max: number): string {
  const s = String(value ?? '').replace(/\s+/g, ' ').trim()
  if (!s) throw new NotificationActionError(400, `${label} is required`)
  if (s.length > max) throw new NotificationActionError(400, `${label} must be ${max} characters or fewer`)
  if (/["\\]/.test(s)) {
    throw new NotificationActionError(400, `${label} can't contain double quotes or backslashes. Use a single quote (') instead.`)
  }
  return s
}

export function parseMessage(
  raw: unknown,
  channels: DeliveryChannel[]
): { templateKey: string; templateData: Record<string, unknown>; category: BroadcastCategory | null } {
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  const m = raw as any

  if (m?.mode === 'template') {
    const t = getTemplate(String(m.templateId ?? ''))
    if (!t) throw new NotificationActionError(400, 'Unknown template')
    if (m.language !== 'en' && m.language !== 'ur') throw new NotificationActionError(400, 'Choose a language')
    const { data, errors } = checkValues(t, m.values ?? {})
    if (errors.length) throw new NotificationActionError(400, errors.join('. '))
    // The category is the template's own. The browser can't override it.
    return { templateKey: templateKey(t.id, m.language), templateData: data, category: t.category }
  }

  if (m?.mode === 'custom') {
    const title = cleanText(m.title, 'Title', LIMITS.title)
    const body = cleanText(m.body, 'Message', channels.includes('sms') ? LIMITS.smsBody : LIMITS.pushBody)
    const deepLink = String(m.deepLink ?? '').trim() || DEFAULT_DEEP_LINK
    if (!DEEP_LINK_RE.test(deepLink)) {
      throw new NotificationActionError(400, 'Deep link must look like tasklync://home')
    }
    const imageUrl = m.imageUrl && typeof m.imageUrl === 'string' && m.imageUrl.trim() ? m.imageUrl.trim() : undefined
    return {
      templateKey: CUSTOM_TEMPLATE_KEY,
      templateData: { title, body, deepLink, ...(imageUrl ? { imageUrl, image: imageUrl } : {}) },
      category: null,
    }
  }

  throw new NotificationActionError(400, 'Write a message first')
}

export function segmentFilterFor(a: AudienceSelection): Record<string, unknown> {
  switch (a.segmentType) {
    case 'category_city':
      return { category_id: a.categoryId, city: a.city }
    case 'custom_ids':
      return a.audience === 'users' ? { user_ids: a.ids } : { worker_ids: a.ids }
    default:
      return {}
  }
}

/** notification-service dedups (recipient, template_key) for 30s. A second launch inside that window would lose messages. */
export async function assertNoRecentDuplicate(templateKeyValue: string) {
  const { rows } = await db.query(
    `SELECT EXTRACT(EPOCH FROM (now() - created_at))::int AS age
     FROM broadcast_campaigns
     WHERE template_key = $1 AND created_at > now() - ($2 || ' seconds')::interval
     ORDER BY created_at DESC LIMIT 1`,
    [templateKeyValue, String(DUPLICATE_LAUNCH_WINDOW_SECONDS)]
  )
  if (rows[0]) {
    const wait = Math.max(1, DUPLICATE_LAUNCH_WINDOW_SECONDS - Number(rows[0].age))
    throw new NotificationActionError(
      429,
      `This template was launched ${rows[0].age}s ago. Wait ${wait} more seconds so the service's duplicate filter doesn't drop messages.`
    )
  }
}

// These mirror the segment queries in notification-service (broadcast-resolve-and-send.job.ts).
// The service is the source of truth: the real count appears as total_recipients once the job starts.

export function segmentSql(sel: AudienceSelection): {
  sql: string
  params: unknown[]
  entity: 'user' | 'worker'
} {
  switch (sel.segmentType) {
    case 'all_users':
      return { sql: `SELECT u.id FROM users u WHERE u.deleted_at IS NULL`, params: [], entity: 'user' }
    case 'all_workers':
      return { sql: `SELECT w.id FROM workers w WHERE w.deleted_at IS NULL`, params: [], entity: 'worker' }
    case 'category_city':
      return {
        sql: `SELECT DISTINCT w.id
              FROM workers w
              JOIN worker_skills ws ON ws.worker_id = w.id
              WHERE w.deleted_at IS NULL
                AND w.verification_status::text = 'VERIFIED'
                AND ws.category_id = $1
                AND w.city = $2`,
        params: [sel.categoryId, sel.city],
        entity: 'worker',
      }
    case 'custom_ids': {
      const table = sel.audience === 'users' ? 'users' : 'workers'
      return {
        sql: `SELECT t.id FROM ${table} t WHERE t.deleted_at IS NULL AND t.id = ANY($1::uuid[])`,
        params: [sel.ids],
        entity: sel.audience === 'users' ? 'user' : 'worker',
      }
    }
  }
}

export async function estimateAudience(
  sel: AudienceSelection,
  category: BroadcastCategory
): Promise<AudienceEstimate> {
  const seg = segmentSql(sel)
  const table = seg.entity === 'user' ? 'users' : 'workers'

  const sampleRes = await db.query(
    `WITH seg AS (${seg.sql})
     SELECT t.name, t.phone FROM seg JOIN ${table} t ON t.id = seg.id
     ORDER BY t.name NULLS LAST LIMIT 5`,
    seg.params
  )
  const sample = sampleRes.rows.map((r) => ({ name: r.name as string | null, phone: r.phone as string | null }))

  // Reach = who will actually get a push/SMS, after the opt-out gates in notification-service:
  //   contact info present AND (global switch, users only) AND category preference
  //   (no preference row => provider default: marketing OFF, platform ON for push; SMS OFF for both).
  const pushDefault = category === 'marketing' ? 'false' : 'true'
  const globalPush = seg.entity === 'user' ? ' AND COALESCE(t.push_enabled, true)' : ''
  const catParam = `$${seg.params.length + 1}`

  try {
    const { rows } = await db.query(
      `WITH seg AS (${seg.sql})
       SELECT
         count(*)::int AS segment_count,
         count(*) FILTER (WHERE t.fcm_token IS NOT NULL${globalPush}
                            AND COALESCE(p.push_enabled, ${pushDefault}))::int AS push_reach,
         count(*) FILTER (WHERE t.phone IS NOT NULL
                            AND COALESCE(p.sms_enabled, false))::int AS sms_reach
       FROM seg
       JOIN ${table} t ON t.id = seg.id
       LEFT JOIN notification_preferences p
         ON p.recipient_id = t.id
        AND p.recipient_type::text = '${seg.entity}'
        AND p.category = ${catParam}`,
      [...seg.params, category]
    )
    const r = rows[0]
    return {
      entity: seg.entity,
      segmentCount: r.segment_count,
      pushReach: r.push_reach,
      smsReach: r.sms_reach,
      sample,
    }
  } catch (err) {
    // fcm_token column or notification_preferences table not there yet: fall back to the plain count.
    const code = (err as { code?: string } | null)?.code
    if (code !== '42703' && code !== '42P01') throw err
    const { rows } = await db.query(`WITH seg AS (${seg.sql}) SELECT count(*)::int AS n FROM seg`, seg.params)
    return { entity: seg.entity, segmentCount: rows[0].n, pushReach: null, smsReach: null, sample }
  }
}

/** Turns pasted phone numbers / emails / UUIDs into recipient IDs. Phones match on the last 10 digits. */
export async function resolveRecipients(
  audience: 'users' | 'workers',
  identifiers: string[]
): Promise<ResolveResult> {
  const table = audience === 'users' ? 'users' : 'workers'
  const uuids: string[] = []
  const phones: string[] = []
  const emails: string[] = []
  const notFound: string[] = []

  const parsed = identifiers.map((raw) => {
    const s = raw.trim()
    if (isUuid(s)) {
      uuids.push(s)
      return { raw, kind: 'id' as const, key: s }
    }
    if (s.includes('@')) {
      emails.push(s.toLowerCase())
      return { raw, kind: 'email' as const, key: s.toLowerCase() }
    }
    const digits = s.replace(/\D/g, '')
    if (digits.length >= 10) {
      phones.push(digits.slice(-10))
      return { raw, kind: 'phone' as const, key: digits.slice(-10) }
    }
    notFound.push(raw)
    return null
  })

  const { rows } = await db.query(
    `SELECT t.id, t.name, t.phone, t.email, t.fcm_token
     FROM ${table} t
     WHERE t.deleted_at IS NULL
       AND (
         t.id = ANY($1::uuid[])
         OR right(regexp_replace(COALESCE(t.phone, ''), '\\D', '', 'g'), 10) = ANY($2::text[])
         OR lower(COALESCE(t.email, '')) = ANY($3::text[])
       )
     LIMIT ${LIMITS.specificPeople}`,
    [uuids, phones, emails]
  )

  const ids = new Set<string>(rows.map((r) => r.id))
  const tails = new Set<string>(
    rows.map((r) => String(r.phone ?? '').replace(/\D/g, '').slice(-10)).filter(Boolean)
  )
  const mails = new Set<string>(rows.map((r) => String(r.email ?? '').toLowerCase()).filter(Boolean))

  for (const p of parsed) {
    if (!p) continue
    const hit = p.kind === 'id' ? ids.has(p.key) : p.kind === 'phone' ? tails.has(p.key) : mails.has(p.key)
    if (!hit) notFound.push(p.raw)
  }

  return {
    matched: rows.map((r) => ({
      id: r.id,
      name: r.name,
      phone: r.phone,
      email: r.email,
      hasFcmToken: Boolean(r.fcm_token),
    })),
    notFound,
  }
}

export function buildCampaignWhere(sp: URLSearchParams) {
  const where: string[] = []
  const params: unknown[] = []
  const push = (v: unknown) => {
    params.push(v)
    return `$${params.length}`
  }

  const search = (sp.get('search') || '').trim()
  const status = (sp.get('status') || 'all').toUpperCase()
  const segmentType = sp.get('segmentType') || 'all'

  if (search) {
    const p = push(`%${search}%`)
    where.push(`(c.title ILIKE ${p} OR c.template_key ILIKE ${p})`)
  }
  if ((BROADCAST_STATUSES as readonly string[]).includes(status)) {
    where.push(`c.status::text = ${push(status)}`)
  }
  if ((SEGMENT_TYPES as readonly string[]).includes(segmentType)) {
    where.push(`c.segment_type::text = ${push(segmentType)}`)
  }
  return { whereSql: where.length ? `WHERE ${where.join(' AND ')}` : '', params }
}

import { NextRequest, NextResponse } from 'next/server'

const ROLES = ['SUPER_ADMIN', 'OPS', 'FINANCE', 'SUPPORT']
const MAX_SUPER_ADMINS = 2

export async function PATCH(
  req: NextRequest,
  { params }: { params: Promise<{ adminId: string }> }
) {
  const guard = await requirePermission('admin_users:manage')
  if (!guard.ok) return guard.response

  const { adminId } = await params
  if (adminId === guard.admin.adminId) {
    return NextResponse.json({ error: 'You cannot change your own role or status' }, { status: 409 })
  }

  const body = await req.json().catch(() => ({}))
  const role: string | undefined = body.role
  const isActive: boolean | undefined = typeof body.isActive === 'boolean' ? body.isActive : undefined
  if (role === undefined && isActive === undefined) {
    return NextResponse.json({ error: 'Nothing to update' }, { status: 400 })
  }
  if (role !== undefined && !ROLES.includes(role)) return NextResponse.json({ error: 'Invalid role' }, { status: 400 })

  const cur = await db.query(`SELECT email, role, is_active FROM admin_users WHERE id = $1`, [adminId])
  if (cur.rows.length === 0) return NextResponse.json({ error: 'Admin not found' }, { status: 404 })
  const before = cur.rows[0]

  const nextRole = role ?? before.role
  const nextActive = isActive ?? before.is_active

  // never leave the platform without a Super Admin, never exceed the cap
  const stillSuper = nextRole === 'SUPER_ADMIN' && nextActive
  if (before.role === 'SUPER_ADMIN' && before.is_active && !stillSuper) {
    const others = await db.query(
      `SELECT count(*)::int AS n FROM admin_users WHERE role = 'SUPER_ADMIN' AND is_active AND id <> $1`,
      [adminId]
    )
    if (others.rows[0].n === 0) {
      return NextResponse.json({ error: 'This is the last active Super Admin' }, { status: 409 })
    }
  }
  if (stillSuper && !(before.role === 'SUPER_ADMIN' && before.is_active)) {
    const c = await db.query(`SELECT count(*)::int AS n FROM admin_users WHERE role = 'SUPER_ADMIN' AND is_active`)
    if (c.rows[0].n >= MAX_SUPER_ADMINS) {
      return NextResponse.json({ error: `At most ${MAX_SUPER_ADMINS} Super Admins are allowed` }, { status: 409 })
    }
  }

  await db.query(`UPDATE admin_users SET role = $2, is_active = $3, updated_at = NOW() WHERE id = $1`, [
    adminId, nextRole, nextActive,
  ])

  // A role or status change must take effect now, not at the next refresh.
  await destroyAllSessions(adminId)
  if (!nextActive) await revokeAllTrustedDevices(adminId)

  await logAdminAction({
    adminId: guard.admin.adminId,
    adminEmail: guard.admin.adminEmail,
    action: !nextActive && before.is_active ? 'admin.deactivated' : 'admin.updated',
    resourceType: 'admin_user',
    resourceId: adminId,
    oldValue: { role: before.role, isActive: before.is_active },
    newValue: { role: nextRole, isActive: nextActive },
    ip: getClientIp(req),
    userAgent: req.headers.get('user-agent') ?? undefined,
  })

import { NextRequest, NextResponse } from 'next/server'
import { db } from '@/lib/db'
import { requirePermission } from '@/lib/rbac-guard'
import { logAdminAction, getClientIp } from '@/lib/audit'
import { isUuid } from '@/lib/workers-api'
import { CANCELLABLE_BY_ADMIN, STATUS_LABEL, type BookingStatus } from '@/lib/booking-constants'
import { bookingNotFound } from '@/lib/bookings-api'
import { bookingSvc } from '@/lib/services/booking'
import { serviceErrorResponse } from '@/lib/internal-client'

export async function POST(
  req: NextRequest,
  { params }: { params: Promise<{ bookingId: string }> }
) {
  const guard = await requirePermission('bookings:cancel')
  if (!guard.ok) return guard.response

  const { bookingId } = await params
  if (!isUuid(bookingId)) return bookingNotFound()

  const body = await req.json().catch(() => ({}))
  const reason = String(body.reason ?? '').trim()
  const refundPercent = Number(body.refundPercent) // COD: percent of the platform fee that is waived

  if (!reason) return NextResponse.json({ error: 'A reason is required' }, { status: 400 })
  if (!Number.isFinite(refundPercent) || refundPercent < 0 || refundPercent > 100) {
    return NextResponse.json({ error: 'The percentage must be between 0 and 100' }, { status: 400 })
  }

  const admin = { id: guard.admin.adminId, email: guard.admin.adminEmail }

  try {
    const cur = await db.query(`SELECT status::text AS status FROM bookings WHERE id = $1`, [bookingId])
    if (cur.rows.length === 0) return bookingNotFound()
    const status = cur.rows[0].status as BookingStatus
    if (!CANCELLABLE_BY_ADMIN.includes(status)) {
      return NextResponse.json(
        { error: `A booking that is ${STATUS_LABEL[status].toLowerCase()} can't be cancelled.` },
        { status: 422 }
      )
    }

    const result = await bookingSvc.cancel(bookingId, admin, { reason, refundPercent })

    await logAdminAction({
      adminId: admin.id,
      adminEmail: admin.email,
      action: 'booking.cancelled_by_admin',
      resourceType: 'booking',
      resourceId: bookingId,
      oldValue: { status },
      newValue: { status: 'CANCELLED', refundPercent, refundAmount: result.refund_amount },
      notes: reason,
      ip: getClientIp(req),
      userAgent: req.headers.get('user-agent') ?? undefined,
    })

    return NextResponse.json({ ok: true, refundPercent, refundAmount: result.refund_amount })
  } catch (err) {
    return serviceErrorResponse(err, '[POST booking cancel]')
  }
}

import { NextRequest, NextResponse } from 'next/server'
import { db } from '@/lib/db'

const SORT_COLUMNS: Record<string, string> = {
  sort_order: 'c.sort_order',
  name: 'c.name',
  active_workers: 'active_worker_count',
}

const ID_SLUG_RE = /^[a-z0-9][a-z0-9-]{1,48}[a-z0-9]$/

export async function GET(req: NextRequest) {
  const guard = await requirePermission('categories:view')
  if (!guard.ok) return guard.response

  const sp = req.nextUrl.searchParams
  const search = (sp.get('search') || '').trim()
  const status = sp.get('status') || 'all'
  const sortBy = SORT_COLUMNS[sp.get('sortBy') || 'sort_order'] || 'c.sort_order'
  const sortDir = sp.get('sortDir') === 'desc' ? 'DESC' : 'ASC'
  const page = Math.max(1, parseInt(sp.get('page') || '1', 10))
  const pageSize = Math.min(100, Math.max(1, parseInt(sp.get('pageSize') || '20', 10)))
  const offset = (page - 1) * pageSize

  const where: string[] = []
  const params: any[] = []
  let i = 1

  if (search) {
    params.push(`%${search}%`)
    where.push(`(c.name ILIKE $${i} OR c.name_ur ILIKE $${i} OR c.id ILIKE $${i})`)
    i++
  }
  if (status !== 'all') {
    params.push(status === 'active')
    where.push(`c.is_active = $${i}`)
    i++
  }

  const whereSql = where.length ? `WHERE ${where.join(' AND ')}` : ''

  // active_worker_count / service_count computed per category via correlated subqueries
  const listSql = `
    SELECT
      c.id, c.name, c.name_ur, c.icon_url, c.parent_id, c.is_active, c.sort_order, c.created_at,
      (SELECT count(*)::int FROM services s WHERE s.category_id = c.id) AS service_count,
      (SELECT count(DISTINCT ws.worker_id)::int
       FROM worker_skills ws
       JOIN workers w ON w.id = ws.worker_id
       WHERE ws.category_id = c.id AND w.verification_status = 'VERIFIED' AND w.is_active = true
      ) AS active_worker_count
    FROM categories c
    ${whereSql}
    ORDER BY ${sortBy} ${sortDir}
    LIMIT $${i} OFFSET $${i + 1}
  `
  const countSql = `SELECT count(*)::int AS total FROM categories c ${whereSql}`
  const listParams = [...params, pageSize, offset]

  try {
    const [listRes, countRes] = await Promise.all([
      db.query(listSql, listParams),
      db.query(countSql, params),
    ])

    const data = listRes.rows.map((r) => ({
      id: r.id,
      name: r.name,
      nameUr: r.name_ur,
      iconUrl: r.icon_url,
      parentId: r.parent_id,
      isActive: r.is_active,
      sortOrder: r.sort_order,
      serviceCount: r.service_count,
      activeWorkerCount: r.active_worker_count,
      createdAt: r.created_at,
    }))
    const total = countRes.rows[0]?.total ?? 0

    return NextResponse.json({
      data, page, pageSize, total,
      totalPages: Math.max(1, Math.ceil(total / pageSize)),
    })
  } catch (err) {
    console.error('[GET /api/admin/categories]', err)
    return NextResponse.json({ error: 'Failed to load categories' }, { status: 500 })
  }
}

  parseChannels,

  segmentFilterFor,
} from '@/lib/notifications-api'

export const dynamic = 'force-dynamic'

export async function POST(req: NextRequest) {
  const guard = await requirePermission('notifications:send')
  if (!guard.ok) return guard.response

  const body = await req.json().catch(() => ({}))

  try {
    const label = String(body.title ?? '').replace(/\s+/g, ' ').trim()
    if (!label) throw new NotificationActionError(400, 'Give this campaign a name')
    if (label.length > LIMITS.label) {
      throw new NotificationActionError(400, `Campaign name must be ${LIMITS.label} characters or fewer`)
    }

    const audience = parseAudience(body.audience)
    if (
      BROAD_SEGMENTS.includes(audience.segmentType) &&
      !hasPermission(guard.admin.role, 'notifications:broadcast-all')
    ) {
      throw new NotificationActionError(403, 'Only a Super Admin can send to all users or all workers.')
    }

    const channels = parseChannels(body.channels)
    const message = parseMessage(body.message, channels)
    const category = message.category ?? parseCategory(body.category)

    await assertNoRecentDuplicate(message.templateKey)

    // notification-service creates the campaign and sends in the background: the answer is 202 for real now.
    const result = await launchBroadcast({
      title: label,
      template_key: message.templateKey,
      template_data: message.templateData,
      channels,
      category,
      segment_type: audience.segmentType,
      segment_filter: segmentFilterFor(audience),
      created_by: guard.admin.adminId,
    })

    await logAdminAction({
      adminId: guard.admin.adminId,
      adminEmail: guard.admin.adminEmail,
      action: 'notification.broadcast_launched',
      resourceType: 'broadcast_campaign',
      resourceId: result.campaign_id,
      newValue: {
        templateKey: message.templateKey,
        category,
        channels,
        segmentType: audience.segmentType,
        recipientsSpecified: audience.segmentType === 'custom_ids' ? audience.ids.length : null,
        acknowledgedQuietHours: body.acknowledgeQuietHours === true,
      },
      notes: label,
      ip: getClientIp(req),
      userAgent: req.headers.get('user-agent') ?? undefined,
    })

    return NextResponse.json({ campaignId: result.campaign_id, status: result.status }, { status: 202 })
  } catch (err) {
    return notificationsErrorResponse(err, '[POST /api/admin/notifications/broadcasts]')
  }

  { key: 'urgencyWithin2hMultiplier', label: 'Urgent, within 2 hours multiplier', hint: 'e.g. 1.5' },
  { key: 'offPeakMultiplier', label: 'Off-peak multiplier', hint: 'Stored for later. Not applied to prices yet.' },
]

export default function PricingSettings() {
  const { config, isLoading, mutate } = usePricingConfig()
  const [form, setForm] = useState<Form | null>(null)
  const [saving, setSaving] = useState(false)

  useEffect(() => {
    if (!config) return
    setForm({
      customerFeePercent: String(config.customerFeePercent),
      workerCommissionPercent: String(config.workerCommissionPercent),
      demandThreshold: String(config.demandThreshold),
      demandMultiplier: String(config.demandMultiplier),
      urgencySameDayMultiplier: String(config.urgencySameDayMultiplier),
      urgencyWithin2hMultiplier: String(config.urgencyWithin2hMultiplier),
      offPeakMultiplier: String(config.offPeakMultiplier),
    })
  }, [config])

  if (isLoading || !form) return <div className="h-40 animate-pulse rounded-lg bg-gray-100" />

  async function save() {
    if (!form) return
    setSaving(true)
    try {
      const customer = Number(form.customerFeePercent)
      const worker = Number(form.workerCommissionPercent)
      await pricingConfigActions.update({
        customerFeePercent: customer,
        workerCommissionPercent: worker,
        platformFeePercent: customer + worker,
        demandThreshold: Number(form.demandThreshold),
        demandMultiplier: Number(form.demandMultiplier),
        urgencySameDayMultiplier: Number(form.urgencySameDayMultiplier),
        urgencyWithin2hMultiplier: Number(form.urgencyWithin2hMultiplier),
        offPeakMultiplier: Number(form.offPeakMultiplier),
      })
      toast.success('Saved. Bookings use the new prices within about a minute.')
      await mutate()
    } catch (e) {
      toast.error((e as Error).message)
    } finally {
      setSaving(false)
    }
  }

  return (
    <div className="rounded-lg border border-gray-200 bg-white p-5">
      <h2 className="text-sm font-semibold text-gray-900">Pricing</h2>
      <p className="mt-1 text-xs text-gray-500">

      </p>

      <div className="mt-4 grid gap-4 sm:grid-cols-2">
        {FIELDS.map((f) => (
          <label key={f.key} className="block text-sm">
            <span className="font-medium text-gray-700">{f.label}</span>
            <input
              type="number"
              step="any"
              value={form[f.key]}
              onChange={(e) => setForm({ ...form, [f.key]: e.target.value })}
              className="mt-1 w-full rounded-md border border-gray-300 px-3 py-2 text-sm"

        ))}

          onClick={save}

  category?: CategoryListItem
  onClose: () => void
  onSaved: () => void
}

export default function CategoryFormDialog({ mode, category, onClose, onSaved }: Props) {
  const [id, setId] = useState(category?.id ?? '')
  const [name, setName] = useState(category?.name ?? '')
  const [nameUr, setNameUr] = useState(category?.nameUr ?? '')
  const [iconUrl, setIconUrl] = useState(category?.iconUrl ?? '')
  const [sortOrder, setSortOrder] = useState(category?.sortOrder ?? 0)
  const [submitting, setSubmitting] = useState(false)

    setSubmitting(true)

      } else {
        await categoryActions.update(category!.id, input)
        toast.success('Category updated')
      }
      onSaved()
    } catch (e: any) {
      setError(e.message)
    } finally {
      setSubmitting(false)
    }
  }

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 px-4">
      <div className="w-full max-w-md rounded-lg border border-gray-200 bg-white p-5 shadow-sm">

          <Input
            label="Category ID"
            placeholder="ac-repair"
            value={id}
            onChange={(e) => setId(e.target.value)}
            disabled={mode === 'edit'}
            helperText={mode === 'create' ? 'Lowercase letters, numbers, hyphens only. Cannot be changed later.' : undefined}
          />
          <Input label="Name" value={name} onChange={(e) => setName(e.target.value)} />
          <Input label="Name (Urdu)" value={nameUr} onChange={(e) => setNameUr(e.target.value)} />
          <Input label="Icon URL" value={iconUrl} onChange={(e) => setIconUrl(e.target.value)} />
          <Input
            label="Sort order"
            type="number"
            value={sortOrder}
            onChange={(e) => setSortOrder(Number(e.target.value))}
          />
        </div>

        {error && <p className="mt-3 text-xs text-red-600">{error}</p>}

        <div className="mt-5 flex justify-end gap-2">

/** Shared submit wrapper: runs the action, resets, closes, surfaces the error. */
function useSubmit(onDone: (result?: any) => void, onClose: () => void) {
  const [submitting, setSubmitting] = useState(false)
  const [error, setError] = useState<string | null>(null)

  async function run(fn: () => Promise<any>) {
    setError(null)
    setSubmitting(true)
    try {
      const result = await fn()
      onClose()
      onDone(result)
    } catch (e: any) {
      setError(e?.message || 'Something went wrong')
    } finally {
      setSubmitting(false)
    }
  }
  return { submitting, error, run, setError }
}

function useDebounced<T>(value: T, ms: number): T {
  const [v, setV] = useState(value)
  useEffect(() => {
    const t = setTimeout(() => setV(value), ms)
    return () => clearTimeout(t)
  }, [value, ms])
  return v
}

// ---------- Stuck detection (kanban + table highlight) ----------

export type StuckLevel = 'ok' | 'amber' | 'red'

type StuckInput = Pick<
  BookingListItem,
  'status' | 'createdAt' | 'updatedAt' | 'scheduledAt' | 'startedAt' | 'durationHours'
>

/**
 * PENDING              amber 20 min / red 30 min since created (auto-cancel is at 30)
 * ACCEPTED             amber 30 min / red 2 h past the scheduled start
 * IN_PROGRESS          amber 1 h / red 3 h past the expected end
 * COMPLETED_BY_WORKER  amber 12 h / red 24 h waiting for the customer
 */
export function stuckLevel(b: StuckInput, now: number = Date.now()): StuckLevel {
  const minsSince = (iso: string) => (now - new Date(iso).getTime()) / 60_000

  switch (b.status) {
    case 'PENDING': {
      const m = minsSince(b.createdAt)
      return m >= 30 ? 'red' : m >= 20 ? 'amber' : 'ok'
    }
    case 'ACCEPTED': {
      const late = minsSince(b.scheduledAt)
      return late >= 120 ? 'red' : late >= 30 ? 'amber' : 'ok'
    }
    case 'IN_PROGRESS': {
      const over = minsSince(b.startedAt ?? b.updatedAt) - b.durationHours * 60
      return over >= 180 ? 'red' : over >= 60 ? 'amber' : 'ok'
    }
    case 'COMPLETED_BY_WORKER': {
      const h = minsSince(b.updatedAt) / 60
      return h >= 24 ? 'red' : h >= 12 ? 'amber' : 'ok'
    }
    default:
      return 'ok'
  }
}

import { usePathname } from 'next/navigation'
import { ChevronRight } from 'lucide-react'
import { NAV_ITEMS } from '@/lib/nav-config'

function humanize(segment: string) {
  // e.g. "workerId" style dynamic segments just get labeled generically
  if (/^[0-9a-fA-F-]{8,}$/.test(segment)) return `#${segment.slice(0, 8)}`
  return segment
    .replace(/-/g, ' ')
    .replace(/\b\w/g, (c) => c.toUpperCase())
}

export function Breadcrumb() {
  const pathname = usePathname()

  if (pathname === '/') {
    return <span className="text-sm font-medium text-gray-700">Dashboard</span>
  }

  const segments = pathname.split('/').filter(Boolean)
  const topLevelMatch = NAV_ITEMS.find((item) => item.href === `/${segments[0]}`)

  return (
    <div className="flex items-center gap-1.5 text-sm">
      <Link href="/" className="text-gray-400 transition-colors duration-150 hover:text-gray-600">
        Dashboard
      </Link>
      {segments.map((segment, i) => {
        const href = '/' + segments.slice(0, i + 1).join('/')
        const isLast = i === segments.length - 1
        const label = i === 0 && topLevelMatch ? topLevelMatch.label : humanize(segment)

        return (
          <span key={href} className="flex items-center gap-1.5">
            <ChevronRight size={14} className="text-gray-300" />
            {isLast ? (
              <span className="font-medium text-gray-700">{label}</span>
            ) : (
              <Link href={href} className="text-gray-400 transition-colors duration-150 hover:text-gray-600">
                {label}
              </Link>
            )}
          </span>
        )
      })}
    </div>
  )
}

export interface ComposeSeed {
  templateId?: string
  language?: TemplateLanguage
}

function initialDraft(seed: ComposeSeed | null): MessageDraft {
  const t = seed?.templateId ? getTemplate(seed.templateId) : undefined
  if (t) {
    return { mode: 'template', templateId: t.id, language: seed?.language ?? 'en', values: defaultValues(t) }
  }
  return { mode: 'custom', title: '', body: '', deepLink: '' }
}

function previewOf(draft: MessageDraft): { title: string; body: string } {
  if (draft.mode === 'custom') return { title: draft.title.trim(), body: draft.body.trim() }
  const t = getTemplate(draft.templateId)
  return t ? previewTemplate(t, draft.language, draft.values) : { title: '', body: '' }
}

function composeProblems(args: {
  audience: AudienceSelection | null
  draft: MessageDraft
  channels: DeliveryChannel[]
  recipientCount: number | null
}): string[] {
  const { audience, draft, channels, recipientCount } = args
  const problems: string[] = []

  if (!audience) problems.push('Choose who should receive this.')
  else if (recipientCount === 0) problems.push('Nobody matches this audience.')

  if (channels.length === 0) problems.push('Choose at least one channel.')

  if (draft.mode === 'template') {
    const t = getTemplate(draft.templateId)
    if (!t) problems.push('Choose a template.')
    else problems.push(...checkValues(t, draft.values).errors)
  } else {
    const bodyMax = channels.includes('sms') ? LIMITS.smsBody : LIMITS.pushBody
    if (!draft.title.trim()) problems.push('Write a title.')
    if (draft.title.length > LIMITS.title) problems.push(`Title is over ${LIMITS.title} characters.`)
    if (!draft.body.trim()) problems.push('Write a message.')
    if (draft.body.length > bodyMax) problems.push(`Message is over ${bodyMax} characters.`)
    if (/["\\]/.test(draft.title + draft.body)) problems.push("Remove double quotes and backslashes (use ').")
    if (draft.deepLink.trim() && !DEEP_LINK_RE.test(draft.deepLink.trim())) {
      problems.push('Deep link must look like tasklync://home.')
    }
  }
  return problems
}

function SectionTitle({ n, children }: { n: number; children: React.ReactNode }) {
  return (
    <h2 className="mb-3 text-sm font-semibold uppercase tracking-wide text-gray-500">
      {n}. {children}
    </h2>
  )
}

export const DEFAULT_BOOKING_FILTERS: BookingFilters = {
  search: '',
  status: 'all',
  city: 'all',
  categoryId: 'all',
  paymentStatus: 'all',
  hasDispute: 'all',
  amountMin: '',
  amountMax: '',
  dateFrom: null,
  dateTo: null,
  sortBy: 'created_at',
  sortDir: 'desc',
  page: 1,
  pageSize: 20,
}

function filterParams(filters: BookingFilters, includeStatus = true): URLSearchParams {
  const qs = new URLSearchParams()
  if (filters.search) qs.set('search', filters.search)
  if (includeStatus && filters.status !== 'all') qs.set('status', filters.status)
  if (filters.city !== 'all') qs.set('city', filters.city)
  if (filters.categoryId !== 'all') qs.set('categoryId', filters.categoryId)
  if (filters.paymentStatus !== 'all') qs.set('paymentStatus', filters.paymentStatus)
  if (filters.hasDispute !== 'all') qs.set('hasDispute', filters.hasDispute)
  if (filters.amountMin) qs.set('amountMin', filters.amountMin)
  if (filters.amountMax) qs.set('amountMax', filters.amountMax)
  if (filters.dateFrom) qs.set('dateFrom', filters.dateFrom)
  if (filters.dateTo) qs.set('dateTo', filters.dateTo)
  return qs
}

function buildQuery(filters: BookingFilters) {
  const qs = filterParams(filters)
  qs.set('sortBy', filters.sortBy)
  qs.set('sortDir', filters.sortDir)
  qs.set('page', String(filters.page))
  qs.set('pageSize', String(filters.pageSize))
  return qs.toString()
}

/** Query string for the CSV export: filters only, no paging/sorting. */
export function buildBookingExportQuery(filters: BookingFilters): string {
  return filterParams(filters).toString()
}

export function useBookingFilters(initial: Partial<BookingFilters> = {}) {
  const [filters, setFilters] = useState<BookingFilters>({ ...DEFAULT_BOOKING_FILTERS, ...initial })

  function update(patch: Partial<BookingFilters>) {
    // any real filter change resets pagination back to page 1
    const resetsPage = Object.keys(patch).some((k) => k !== 'page')
    setFilters((prev) => ({ ...prev, ...patch, page: resetsPage ? 1 : (patch.page ?? prev.page) }))
  }

  function reset() {
    setFilters(DEFAULT_BOOKING_FILTERS)
  }

  return { filters, update, reset }
}

async function send(method: 'POST' | 'PATCH', url: string, body?: object) {
  const res = await fetch(url, {
    method,
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body ?? {}),
  })
  const json = await res.json().catch(() => ({}))
  if (!res.ok) throw new Error(json.error || 'Action failed')
  return json
}

function toQs(filters: object): string {
  const qs = new URLSearchParams()
  for (const [k, v] of Object.entries(filters as Record<string, string | number | null | undefined>)) {
    if (v === null || v === undefined || v === '' || v === 'all') continue
    qs.set(k, String(v))
  }
  return qs.toString()
}

/** Debounced reach estimate for the composer. */
export function useAudienceEstimate(audience: AudienceSelection | null, category: BroadcastCategory) {
  const [state, setState] = useState<{
    estimate: AudienceEstimate | null
    loading: boolean
    error: string | null
  }>({ estimate: null, loading: false, error: null })

  const key = audience ? JSON.stringify({ audience, category }) : null

  useEffect(() => {
    if (!key) {
      setState({ estimate: null, loading: false, error: null })
      return
    }
    const ctrl = new AbortController()
    setState((s) => ({ ...s, loading: true, error: null }))

    const timer = setTimeout(async () => {
      try {
        const res = await fetch('/api/admin/notifications/audience/estimate', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: key,
          signal: ctrl.signal,
        })
        const json = await res.json().catch(() => ({}))
        if (!res.ok) throw new Error(json.error || 'Could not estimate reach')
        setState({ estimate: json, loading: false, error: null })
      } catch (e) {
        if ((e as Error).name === 'AbortError') return
        setState({ estimate: null, loading: false, error: (e as Error).message })
      }
    }, 400)

    return () => {
      clearTimeout(timer)
      ctrl.abort()
    }
  }, [key])

  return state
}

async function post<T>(url: string, body: object): Promise<T> {
  const res = await fetch(url, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body),
  })
  const json = await res.json().catch(() => ({}))
  if (!res.ok) throw new Error(json.error || 'Action failed')
  return json as T
}

// scripts/generate-broadcast-templates.ts
// Usage:  npx tsx scripts/generate-broadcast-templates.ts <path-to>/notification-service/src/templates
// With no argument it writes to ./generated-templates so you can review first.
import { mkdirSync, writeFileSync } from 'node:fs'
import { join, resolve } from 'node:path'
import { CATALOG, templateKey, type TemplateLanguage } from '../lib/notification-catalog'
import { CUSTOM_TEMPLATE_KEY } from '../lib/notification-constants'

const out = resolve(process.argv[2] ?? './generated-templates')
const pushDir = join(out, 'push')
const smsDir = join(out, 'sms')
mkdirSync(pushDir, { recursive: true })
mkdirSync(smsDir, { recursive: true })

// {{x}} -> {{{x}}} (no HTML escaping inside push/SMS text)
const raw = (s: string) => s.replace(/\{\{\s*(\w+)\s*\}\}/g, '{{{$1}}}')
const writeJson = (file: string, data: object) =>
  writeFileSync(file, JSON.stringify(data, null, 2) + '\n', 'utf8')

let files = 0

for (const t of CATALOG) {
  for (const lang of Object.keys(t.copy) as TemplateLanguage[]) {
    const key = templateKey(t.id, lang)
    const c = t.copy[lang]
    writeJson(join(pushDir, `${key}.json`), {
      title: raw(c.title),
      body: raw(c.body),
      deep_link: t.deepLink,
    })
    writeFileSync(join(smsDir, `${key}.hbs`), `TaskLync: ${raw(c.body)}\n`, 'utf8')
    files += 2
  }
}

// Generic template for admin-written messages.
writeJson(join(pushDir, `${CUSTOM_TEMPLATE_KEY}.json`), {
  title: '{{{title}}}',
  body: '{{{body}}}',
  deep_link: '{{{deepLink}}}',
})
writeFileSync(join(smsDir, `${CUSTOM_TEMPLATE_KEY}.hbs`), 'TaskLync: {{{body}}}\n', 'utf8')
files += 2

console.log(`Wrote ${files} template files to ${out}`)
console.log('Copy push/ and sms/ into services/notification-service/src/templates and restart the service.')
