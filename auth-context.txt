import { NextRequest, NextResponse } from 'next/server'
import { jwtVerify } from 'jose'
import { permissionForPath, hasPermission } from '@/lib/auth/rbac'
import type { AdminRole } from '@/types/auth'

const secret = new TextEncoder().encode(process.env.ADMIN_JWT_SECRET!)

export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl
  const isAuthRoute = pathname.startsWith('/auth')
  const isAuthApi = pathname.startsWith('/api/admin/auth')

  if (isAuthRoute || isAuthApi) return NextResponse.next()

  const accessToken = request.headers.get('authorization')?.replace('Bearer ', '')
    ?? request.cookies.get('admin_access')?.value

  if (!accessToken) {
    return NextResponse.redirect(new URL('/auth/login', request.url))
  }

  try {
    const { payload } = await jwtVerify(accessToken, secret)
    const role = payload.role as AdminRole

    const requiredPermission = permissionForPath(pathname)
    if (requiredPermission && !hasPermission(role, requiredPermission)) {
      return NextResponse.redirect(new URL('/', request.url)) // bounce to dashboard, not an error page
    }

    return NextResponse.next()
  } catch {
    // Invalid, tampered, or expired token â€” never let the request through.
    // Client-side useAdminSession handles silent refresh for API calls;
    // a plain page navigation with a dead token must go straight to login.
    return NextResponse.redirect(new URL('/auth/login', request.url))
  }
}

export const config = {
  matcher: ['/((?!_next/static|_next/image|favicon.ico).*)'],
}

'use client'

import { useState } from 'react'
import { useRouter } from 'next/navigation'

export function LoginForm() {
  const router = useRouter()
  const [email, setEmail] = useState('')
  const [password, setPassword] = useState('')
  const [error, setError] = useState<string | null>(null)
  const [loading, setLoading] = useState(false)

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault()
    setError(null)
    setLoading(true)
    try {
      const res = await fetch('/api/admin/auth/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password }),
      })

      let data: any = null
      try {
        data = await res.json()
      } catch {
        throw new Error('Server error. Please try again in a moment.')
      }

      if (!res.ok) {
        setError(data?.error ?? 'Something went wrong')
        return
      }
      if (data.requiresMFA) {
        sessionStorage.setItem('mfa_temp_token', data.tempToken)
        router.push('/auth/mfa')
        return
      }
      sessionStorage.setItem('admin_access_token', data.accessToken)
      router.push(data.mfaSetupRequired ? '/auth/mfa-setup' : '/')
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Network error. Please try again.')
    } finally {
      setLoading(false)
    }
  }

  return (
    <form onSubmit={handleSubmit} className="space-y-4 bg-bg-surface border border-border rounded-lg p-6 w-full max-w-sm">
      <h1 className="text-lg font-semibold text-text-primary">Admin Sign In</h1>

      <div>
        <label htmlFor="email" className="text-xs font-medium text-gray-700 block mb-1">
          Email
        </label>
        <input
          id="email"
          type="email"
          value={email}
          onChange={(e) => setEmail(e.target.value)}
          required
          autoComplete="email"
          disabled={loading}
          className="h-9 w-full px-3 text-sm border border-gray-300 rounded-md focus:outline-none focus:ring-2 focus:ring-accent disabled:opacity-60 disabled:cursor-not-allowed"
        />
      </div>

      <div>
        <label htmlFor="password" className="text-xs font-medium text-gray-700 block mb-1">
          Password
        </label>
        <input
          id="password"
          type="password"
          value={password}
          onChange={(e) => setPassword(e.target.value)}
          required
          autoComplete="current-password"
          disabled={loading}
          className="h-9 w-full px-3 text-sm border border-gray-300 rounded-md focus:outline-none focus:ring-2 focus:ring-accent disabled:opacity-60 disabled:cursor-not-allowed"
        />
      </div>

      {error && (
        <p role="alert" className="text-xs text-danger">
          {error}
        </p>
      )}

      <button
        type="submit"
        disabled={loading}
        className="cursor-pointer h-9 w-full text-sm font-medium rounded-md bg-accent text-white transition-colors duration-150 hover:bg-accent-hover disabled:opacity-60 disabled:cursor-not-allowed flex items-center justify-center gap-2"

          <span

          />
        )}
        {loading ? 'Signing inâ€¦' : 'Sign in'}
      </button>
    </form>
  )
}

export function MfaForm() {

  const [loading, setLoading] = useState(false)

    e.preventDefault()
    setError(null)
    setLoading(true)
    try {
      const tempToken = sessionStorage.getItem('mfa_temp_token')
      if (!tempToken) {
        setError('Your session expired. Please sign in again.')
        router.push('/auth/login')
        return
      }

      const res = await fetch('/api/admin/auth/mfa/verify', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ tempToken, code }),

      sessionStorage.setItem('admin_access_token', data.accessToken)

      setError(err instanceof Error ? err.message : 'Network error. Please try again.')

      setLoading(false)

'use client'

import { useEffect, useState } from 'react'
import { useRouter } from 'next/navigation'
import { Copy, Check } from 'lucide-react'

export function MfaSetupForm() {
  const router = useRouter()
  const [step, setStep] = useState<'loading' | 'scan' | 'confirmed' | 'setup-error'>('loading')
  const [qrCodeDataUrl, setQrCodeDataUrl] = useState('')
  const [secret, setSecret] = useState('')
  const [code, setCode] = useState('')
  const [backupCodes, setBackupCodes] = useState<string[]>([])
  const [error, setError] = useState<string | null>(null)
  const [confirming, setConfirming] = useState(false)
  const [copied, setCopied] = useState(false)

  useEffect(() => {
    let cancelled = false

    async function loadSetup() {
      try {
        const token = sessionStorage.getItem('admin_access_token')
        if (!token) {
          router.push('/auth/login')
          return
        }

        const res = await fetch('/api/admin/auth/mfa/setup', {
          method: 'POST',
          headers: { Authorization: `Bearer ${token}` },
        })

        let data: any = null
        try {
          data = await res.json()
        } catch {
          throw new Error('Server error. Please refresh and try again.')
        }

        if (!res.ok) {
          throw new Error(data?.error ?? 'Could not start MFA setup.')
        }

        if (cancelled) return
        setQrCodeDataUrl(data.qrCodeDataUrl)
        setSecret(data.secret)
        setStep('scan')
      } catch (err) {
        if (cancelled) return
        setError(err instanceof Error ? err.message : 'Network error. Please try again.')
        setStep('setup-error')
      }
    }

    loadSetup()
    return () => {
      cancelled = true
    }
  }, [router])

  async function handleConfirm(e: React.FormEvent) {
    e.preventDefault()
    setError(null)
    setConfirming(true)
    try {
      const token = sessionStorage.getItem('admin_access_token')
      const res = await fetch('/api/admin/auth/mfa/setup-confirm', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
        body: JSON.stringify({ code }),
      })

      let data: any = null
      try {
        data = await res.json()
      } catch {
        throw new Error('Server error. Please try again.')
      }

      if (!res.ok) {
        setError(data?.error ?? 'Incorrect code')
        return
      }
      setBackupCodes(data.backupCodes)
      setStep('confirmed')
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Network error. Please try again.')
    } finally {
      setConfirming(false)
    }
  }

  async function handleCopyBackupCodes() {
    try {
      await navigator.clipboard.writeText(backupCodes.join('\n'))
      setCopied(true)
      setTimeout(() => setCopied(false), 2000)
    } catch {
      setError('Could not copy automatically. Please select and copy manually.')
    }
  }

  if (step === 'loading') {
    return (
      <div className="space-y-2 bg-bg-surface border border-border rounded-lg p-6 w-full max-w-sm text-center">
        <span
          aria-hidden="true"
          className="inline-block h-4 w-4 rounded-full border-2 border-gray-300 border-t-accent animate-spin"
        />
        <p className="text-sm text-text-secondary">Setting up two factor authenticationâ€¦</p>
      </div>
    )
  }

  if (step === 'setup-error') {
    return (
      <div className="space-y-4 bg-bg-surface border border-border rounded-lg p-6 w-full max-w-sm">
        <h1 className="text-lg font-semibold text-text-primary">Couldn't Start Setup</h1>
        <p role="alert" className="text-xs text-danger">
          {error}
        </p>
        <button
          onClick={() => window.location.reload()}
          className="h-9 w-full text-sm font-medium rounded-md bg-accent text-white transition-colors duration-150 hover:bg-accent-hover"
        >
          Try again
        </button>
      </div>
    )
  }

  if (step === 'confirmed') {
    return (
      <div className="space-y-4 bg-bg-surface border border-border rounded-lg p-6 w-full max-w-sm">
        <h1 className="text-lg font-semibold text-text-primary">Save Your Backup Codes</h1>
        <p className="text-xs text-text-secondary">
          Each code works once, if you lose access to your authenticator app. Store these somewhere safe they won't be shown again.
        </p>
        <div className="relative bg-gray-50 border border-gray-200 rounded-md p-3 pr-10 font-mono text-sm space-y-1">
          {backupCodes.map((c) => (
            <div key={c}>{c}</div>
          ))}
          <button
            type="button"
            onClick={handleCopyBackupCodes}
            title={copied ? 'Copied!' : 'Copy all codes'}
            className="absolute top-2 right-2 h-7 w-7 flex items-center justify-center rounded-md border border-gray-300 bg-white text-gray-600 hover:bg-gray-50 transition-colors duration-150"
          >
            {copied ? <Check size={14} className="text-success" /> : <Copy size={14} />}
          </button>
        </div>

        {copied && <p className="text-xs text-success cursor-pointer">Copied to clipboard.</p>}

        <button
          onClick={() => router.push('/')}
          className="cursor-pointer h-9 w-full text-sm font-medium rounded-md bg-accent text-white transition-colors duration-150 hover:bg-accent-hover"
        >
          Saved these, Continue to Dashboard
        </button>
      </div>
    )
  }

  return (
    <form onSubmit={handleConfirm} className="space-y-4 bg-bg-surface border border-border rounded-lg p-6 w-full max-w-sm">
      <h1 className="text-lg font-semibold text-text-primary">Set Up Two Factor Authentication</h1>
      <p className="text-xs text-text-secondary">Scan this with Google Authenticator or a similar app.</p>

      {qrCodeDataUrl && <img src={qrCodeDataUrl} alt="MFA QR code" className="mx-auto" width={180} height={180} />}

      <p className="text-xs text-text-muted text-center">
        Can't scan? Enter manually: <span className="font-mono">{secret}</span>
      </p>

      <div>
        <label htmlFor="mfa-setup-code" className="text-xs font-medium text-gray-700 block mb-1">
          Enter the 6 digit code to confirm
        </label>
        <input
          id="mfa-setup-code"
          value={code}
          onChange={(e) => setCode(e.target.value)}
          maxLength={6}
          required
          autoFocus
          disabled={confirming}
          className="h-9 w-full px-3 text-sm border border-gray-300 rounded-md font-mono tracking-widest focus:outline-none focus:ring-2 focus:ring-accent disabled:opacity-60 disabled:cursor-not-allowed"
        />
      </div>

      {error && (
        <p role="alert" className="text-xs text-danger">
          {error}
        </p>
      )}

      <button
        type="submit"
        disabled={confirming}
        className="h-9 w-full text-sm font-medium rounded-md bg-accent text-white transition-colors duration-150 hover:bg-accent-hover disabled:opacity-60 disabled:cursor-not-allowed flex items-center justify-center gap-2"
      >
        {confirming && (
          <span
            aria-hidden="true"
            className="h-3.5 w-3.5 rounded-full border-2 border-white/40 border-t-white animate-spin"
          />
        )}
        {confirming ? 'Confirmingâ€¦' : 'Confirm & Enable'}
      </button>
    </form>
  )
}

'use client'

import { useEffect, useState, useCallback } from 'react'
import { useRouter } from 'next/navigation'
import type { AdminUser } from '@/types/auth'

export function useAdminSession() {
  const router = useRouter()
  const [admin, setAdmin] = useState<AdminUser | null>(null)
  const [loading, setLoading] = useState(true)

  const fetchWithAuth = useCallback(async (input: RequestInfo, init: RequestInit = {}) => {
    let token = sessionStorage.getItem('admin_access_token')
    let res = await fetch(input, { ...init, headers: { ...init.headers, Authorization: `Bearer ${token}` } })

    if (res.status === 401) {
      const refreshRes = await fetch('/api/admin/auth/refresh', { method: 'POST' })
      if (!refreshRes.ok) {
        router.push('/auth/login')
        throw new Error('Session expired')
      }
      const { accessToken } = await refreshRes.json()
      sessionStorage.setItem('admin_access_token', accessToken)
      res = await fetch(input, { ...init, headers: { ...init.headers, Authorization: `Bearer ${accessToken}` } })
    }
    return res
  }, [router])

  useEffect(() => {
    fetchWithAuth('/api/admin/auth/session')
      .then((r) => r.json())
      .then(setAdmin)
      .catch(() => setAdmin(null))
      .finally(() => setLoading(false))
  }, [fetchWithAuth])

  return { admin, loading, fetchWithAuth }
}

import { SignJWT, jwtVerify } from 'jose'
import type { AccessTokenPayload, MfaTempTokenPayload } from '@/types/auth'

const secret = new TextEncoder().encode(process.env.ADMIN_JWT_SECRET!)

export async function signAccessToken(payload: Omit<AccessTokenPayload, 'type'>) {
  const ttl = Number(process.env.ADMIN_ACCESS_TOKEN_TTL ?? 900)
  return new SignJWT({ ...payload, type: 'access' })
    .setProtectedHeader({ alg: 'HS256' })
    .setIssuedAt()
    .setExpirationTime(Math.floor(Date.now() / 1000) + ttl)
    .sign(secret)
}

export async function verifyAccessToken(token: string): Promise<AccessTokenPayload> {
  const { payload } = await jwtVerify(token, secret)
  if (payload.type !== 'access') throw new Error('Invalid token type')
  return payload as unknown as AccessTokenPayload
}

export async function signMfaTempToken(adminId: string) {
  const ttl = Number(process.env.ADMIN_MFA_TEMP_TOKEN_TTL ?? 900)
  return new SignJWT({ sub: adminId, purpose: 'mfa_pending', type: 'mfa_temp' })
    .setProtectedHeader({ alg: 'HS256' })
    .setIssuedAt()
    .setExpirationTime(Math.floor(Date.now() / 1000) + ttl)
    .sign(secret)
}

export async function verifyMfaTempToken(token: string): Promise<MfaTempTokenPayload> {
  const { payload } = await jwtVerify(token, secret)
  if (payload.type !== 'mfa_temp') throw new Error('Invalid token type')
  return payload as unknown as MfaTempTokenPayload
}

import { redis } from '@/lib/redis'
import crypto from 'crypto'
import type { AdminRole, SessionRecord } from '@/types/auth'

const REFRESH_TTL = Number(process.env.ADMIN_REFRESH_TOKEN_TTL ?? 28800)
const MAX_SESSIONS = Number(process.env.ADMIN_MAX_CONCURRENT_SESSIONS ?? 2)

function sessionKey(refreshToken: string) {
  return `admin:session:${refreshToken}`
}
function sessionSetKey(adminId: string) {
  return `admin:sessions:${adminId}`
}

export async function createSession(
  adminId: string,
  email: string,
  role: AdminRole,
  ip: string
): Promise<string> {
  const refreshToken = crypto.randomBytes(32).toString('hex')
  const record: SessionRecord = { adminId, email, role, createdAt: new Date().toISOString(), ip }

  await redis.set(sessionKey(refreshToken), JSON.stringify(record), 'EX', REFRESH_TTL)

  // Track sessions per admin, evict oldest if over the concurrent-session limit
  const setKey = sessionSetKey(adminId)
  await redis.zadd(setKey, Date.now(), refreshToken)
  const count = await redis.zcard(setKey)
  if (count > MAX_SESSIONS) {
    const toEvict = await redis.zrange(setKey, 0, count - MAX_SESSIONS - 1)
    if (toEvict.length) {
      await redis.del(...toEvict.map(sessionKey))
      await redis.zrem(setKey, ...toEvict)
    }
  }
  await redis.expire(setKey, REFRESH_TTL)

  return refreshToken
}

export async function getSession(refreshToken: string): Promise<SessionRecord | null> {
  const raw = await redis.get(sessionKey(refreshToken))
  return raw ? JSON.parse(raw) : null
}

export async function rotateSession(oldRefreshToken: string): Promise<string | null> {
  const record = await getSession(oldRefreshToken)
  if (!record) return null

  await redis.del(sessionKey(oldRefreshToken))
  await redis.zrem(sessionSetKey(record.adminId), oldRefreshToken)

  return createSession(record.adminId, record.email, record.role, record.ip)
}

export async function destroySession(refreshToken: string): Promise<void> {
  const record = await getSession(refreshToken)
  await redis.del(sessionKey(refreshToken))
  if (record) await redis.zrem(sessionSetKey(record.adminId), refreshToken)
}

// lib/rbac-guard.ts
import { NextResponse } from 'next/server'
import { cookies } from 'next/headers'
import { verifyAccessToken } from '@/lib/auth/jwt'
import { hasPermission } from '@/lib/auth/rbac'
import type { AdminRole } from '@/types/auth'

interface AdminContext {
  adminId: string
  adminEmail: string
  role: AdminRole
  sessionId: string
}

type PermissionResult =
  | { ok: true; admin: AdminContext }
  | { ok: false; response: NextResponse }

export async function requirePermission(permission: string): Promise<PermissionResult> {
  const cookieStore = await cookies()
  const token = cookieStore.get('admin_access')?.value

  if (!token) {
    return { ok: false, response: NextResponse.json({ error: 'Unauthorized' }, { status: 401 }) }
  }

  let payload
  try {
    payload = await verifyAccessToken(token)
  } catch {
    return { ok: false, response: NextResponse.json({ error: 'Unauthorized' }, { status: 401 }) }
  }

  if (!hasPermission(payload.role, permission)) {
    return { ok: false, response: NextResponse.json({ error: 'Forbidden' }, { status: 403 }) }
  }

  return {
    ok: true,
    admin: {
      adminId: payload.sub,
      adminEmail: payload.email,
      role: payload.role,
      sessionId: payload.sessionId,
    },
  }
}

// lib/parse-user-agent.ts
import { UAParser } from 'ua-parser-js'

export interface ParsedUserAgent {
  isBot: boolean
  deviceType: string
  browser: string
  os: string
}

// Catches your own Postman/curl/internal-tooling traffic so it doesn't
// pollute "real visitor" device/browser stats.
const BOT_PATTERN = /bot|crawler|spider|postman|curl|python-requests|axios|okhttp|insomnia/i

export function parseUserAgent(userAgent: string | null): ParsedUserAgent {
  if (!userAgent) {
    return { isBot: false, deviceType: 'Unknown', browser: 'Unknown', os: 'Unknown' }
  }

  if (BOT_PATTERN.test(userAgent)) {
    return { isBot: true, deviceType: 'Bot / API client', browser: 'N/A', os: 'N/A' }
  }

  const result = new UAParser(userAgent).getResult()

  const deviceType = result.device.type
    ? result.device.type.charAt(0).toUpperCase() + result.device.type.slice(1)
    : 'Desktop' // ua-parser-js leaves type undefined for normal desktop UAs

  return {
    isBot: false,
    deviceType,
    browser: result.browser.name ?? 'Unknown',
    os: result.os.name ?? 'Unknown',
  }
}

import { createInterface } from 'readline/promises'

async function prompt() {
  const rl = createInterface({ input, output })
  try {
    const email = (await rl.question('Super admin email: ')).trim()
    const fullName = (await rl.question('Full name: ')).trim()
    const password = await rl.question(
      'Temporary password (change after first login): '
    )
    return { email, fullName, password }
  } finally {
    rl.close()
  }
}

async function main() {
  const { hashPassword } = await import('../lib/auth/password')
  const { db } = await import('../lib/db')

  const { rows } = await db.query(
    `SELECT COUNT(*)::int AS count FROM admin_users WHERE role = 'SUPER_ADMIN'`
  )
  if (rows[0].count >= 2) {
    console.error('Maximum of 2 super admins already exist.')
    console.error('Additional admins must be created through the admin panel.')
    return
  }

  const { email, fullName, password } = await prompt()

  if (!email || !fullName || !password) {
    console.error('Email, full name, and password are required.')
    return
  }

  const existing = await db.query(
    `SELECT id FROM admin_users WHERE LOWER(email) = LOWER($1) LIMIT 1`,
    [email]
  )
  if (existing.rows.length > 0) {
    console.error(`An admin account with ${email} already exists.`)
    return
  }

  const hash = await hashPassword(password)

  await db.query(
    `INSERT INTO admin_users (email, password_hash, full_name, role, is_active)
     VALUES ($1, $2, $3, 'SUPER_ADMIN', TRUE)`,
    [email, hash, fullName]
  )

  console.log(`\nSuper admin ${email} created successfully.`)
  console.log('MFA setup will be required on first login.')
}

export const runtime = 'nodejs'

import { NextRequest, NextResponse } from 'next/server'
import { z } from 'zod'
import { db } from '@/lib/db'
import { verifyPassword } from '@/lib/auth/password'
import { signAccessToken, signMfaTempToken } from '@/lib/auth/jwt'
import { createSession } from '@/lib/auth/session'
import { logAdminAction } from '@/lib/audit'
import { setAuthCookies } from '@/lib/auth/cookies'

const schema = z.object({
  email: z.string().email(),
  password: z.string().min(1),
})

const MAX_ATTEMPTS = Number(process.env.ADMIN_MAX_FAILED_ATTEMPTS ?? 5)
const LOCKOUT_MINUTES = Number(process.env.ADMIN_LOCKOUT_MINUTES ?? 30)

export async function POST(req: NextRequest) {
  const ip = req.headers.get('x-forwarded-for') ?? 'unknown'
  const body = schema.safeParse(await req.json())
  if (!body.success) {
    return NextResponse.json({ error: 'Invalid email or password format' }, { status: 400 })
  }
  const { email, password } = body.data

  const { rows } = await db.query('SELECT * FROM admin_users WHERE email = $1', [email])
  const admin = rows[0]

  // Deliberately identical error for "no account" and "wrong password" â€” don't leak which one
  const genericError = { error: 'Invalid email or password' }

  if (!admin || !admin.is_active) {
    return NextResponse.json(genericError, { status: 401 })
  }

  if (admin.locked_until && new Date(admin.locked_until) > new Date()) {
    return NextResponse.json(
      { error: `Account locked. Try again after ${new Date(admin.locked_until).toLocaleTimeString()}.` },
      { status: 423 }
    )
  }

  const valid = await verifyPassword(password, admin.password_hash)

  if (!valid) {
    const attempts = admin.failed_attempts + 1
    const lockUntil = attempts >= MAX_ATTEMPTS
      ? new Date(Date.now() + LOCKOUT_MINUTES * 60_000)
      : null

    await db.query(
      'UPDATE admin_users SET failed_attempts = $1, locked_until = $2 WHERE id = $3',
      [attempts, lockUntil, admin.id]
    )
    await logAdminAction({
      adminId: admin.id,
      adminEmail: admin.email,
      action: 'auth.login_failed',
      ip,
      notes: `Failed attempt ${attempts}/${MAX_ATTEMPTS}`,
    })

    return NextResponse.json(genericError, { status: 401 })
  }

  // Success â€” reset failed attempts
  await db.query(
    'UPDATE admin_users SET failed_attempts = 0, locked_until = NULL, last_login_at = NOW(), last_login_ip = $1 WHERE id = $2',
    [ip, admin.id]
  )

  if (admin.mfa_enabled) {
    const tempToken = await signMfaTempToken(admin.id)
    await logAdminAction({
      adminId: admin.id, adminEmail: admin.email, action: 'auth.login_mfa_pending', ip,
    })
    return NextResponse.json({ requiresMFA: true, tempToken })
  }

  // No MFA set up yet â€” issue full session (MFA setup is enforced on first login, see below)
  const refreshToken = await createSession(admin.id, admin.email, admin.role, ip)
  const accessToken = await signAccessToken({
    sub: admin.id, email: admin.email, role: admin.role, sessionId: refreshToken,
  })

  await logAdminAction({ adminId: admin.id, adminEmail: admin.email, action: 'auth.login_success', ip })

  const res = NextResponse.json({
    accessToken,
    admin: { id: admin.id, email: admin.email, fullName: admin.full_name, role: admin.role, mfaEnabled: false },
    mfaSetupRequired: admin.role === 'SUPER_ADMIN', // mandatory for SUPER_ADMIN per your spec
  })
  setAuthCookies(res, accessToken, refreshToken)
  return res
}

  if (refreshToken) {

  return res

export const runtime = 'nodejs'

import { NextRequest, NextResponse } from 'next/server'
import { z } from 'zod'
import { db } from '@/lib/db'
import { verifyAccessToken } from '@/lib/auth/jwt'
import { verifyTotp, generateBackupCodes, hashBackupCodes } from '@/lib/auth/mfa'
import { redis } from '@/lib/redis'
import { logAdminAction } from '@/lib/audit'

const schema = z.object({ code: z.string().min(6) })

export async function POST(req: NextRequest) {
  const ip = req.headers.get('x-forwarded-for') ?? 'unknown'
  const auth = req.headers.get('authorization')?.replace('Bearer ', '')
  if (!auth) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  const payload = await verifyAccessToken(auth).catch(() => null)
  if (!payload) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const body = schema.safeParse(await req.json())
  if (!body.success) return NextResponse.json({ error: 'Invalid code' }, { status: 400 })

  const pendingSecret = await redis.get(`admin:mfa_pending:${payload.sub}`)
  if (!pendingSecret) {
    return NextResponse.json({ error: 'Setup session expired, start again' }, { status: 400 })
  }

  if (!verifyTotp(body.data.code, pendingSecret)) {
    return NextResponse.json({ error: 'Incorrect code' }, { status: 400 })
  }

  await db.query('UPDATE admin_users SET mfa_secret = $1, mfa_enabled = TRUE WHERE id = $2', [
    pendingSecret, payload.sub,
  ])
  await redis.del(`admin:mfa_pending:${payload.sub}`)

  const backupCodes = generateBackupCodes(10)
  const hashes = await hashBackupCodes(backupCodes)
  await db.query('DELETE FROM admin_mfa_backup_codes WHERE admin_id = $1', [payload.sub])
  for (const hash of hashes) {
    await db.query('INSERT INTO admin_mfa_backup_codes (admin_id, code_hash) VALUES ($1, $2)', [
      payload.sub, hash,
    ])
  }

  await logAdminAction({ adminId: payload.sub, adminEmail: payload.email, action: 'auth.mfa_enabled', ip })

  // Backup codes are shown to the admin exactly once â€” plaintext, never stored, never logged
  return NextResponse.json({ backupCodes })
}

export const runtime = 'nodejs'

import { NextRequest, NextResponse } from 'next/server'
import { z } from 'zod'
import { db } from '@/lib/db'
import { verifyMfaTempToken, signAccessToken } from '@/lib/auth/jwt'
import { verifyTotp, verifyBackupCode } from '@/lib/auth/mfa'
import { createSession } from '@/lib/auth/session'
import { logAdminAction } from '@/lib/audit'
import { setAuthCookies } from '@/lib/auth/cookies'

const schema = z.object({
  tempToken: z.string(),
  code: z.string().min(6),
})

export async function POST(req: NextRequest) {
  const ip = req.headers.get('x-forwarded-for') ?? 'unknown'
  const body = schema.safeParse(await req.json())
  if (!body.success) return NextResponse.json({ error: 'Invalid request' }, { status: 400 })

  let adminId: string
  try {
    const payload = await verifyMfaTempToken(body.data.tempToken)
    adminId = payload.sub
  } catch {
    return NextResponse.json({ error: 'MFA session expired, please log in again' }, { status: 401 })
  }

  const { rows } = await db.query('SELECT * FROM admin_users WHERE id = $1', [adminId])
  const admin = rows[0]
  if (!admin) return NextResponse.json({ error: 'Account not found' }, { status: 404 })

  const totpValid = admin.mfa_secret ? verifyTotp(body.data.code, admin.mfa_secret) : false

  let usedBackupCode = false
  if (!totpValid) {
    const { rows: backupRows } = await db.query(
      'SELECT id, code_hash FROM admin_mfa_backup_codes WHERE admin_id = $1 AND used_at IS NULL',
      [adminId]
    )
    const idx = await verifyBackupCode(body.data.code, backupRows.map((r) => r.code_hash))
    if (idx === -1) {
      await logAdminAction({ adminId, adminEmail: admin.email, action: 'auth.mfa_failed', ip })
      return NextResponse.json({ error: 'Invalid code' }, { status: 401 })
    }
    usedBackupCode = true
    await db.query('UPDATE admin_mfa_backup_codes SET used_at = NOW() WHERE id = $1', [backupRows[idx].id])
  }

  const refreshToken = await createSession(admin.id, admin.email, admin.role, ip)
  const accessToken = await signAccessToken({
    sub: admin.id, email: admin.email, role: admin.role, sessionId: refreshToken,
  })

  await logAdminAction({
    adminId: admin.id, adminEmail: admin.email, ip,
    action: usedBackupCode ? 'auth.mfa_success_backup_code' : 'auth.mfa_success',
  })

  const res = NextResponse.json({
    accessToken,
    admin: { id: admin.id, email: admin.email, fullName: admin.full_name, role: admin.role, mfaEnabled: true },
  })
  setAuthCookies(res, accessToken, refreshToken)
  return res
}

export const runtime = 'nodejs'

import { NextRequest, NextResponse } from 'next/server'
import { rotateSession, getSession } from '@/lib/auth/session'
import { signAccessToken } from '@/lib/auth/jwt'

export async function POST(req: NextRequest) {
  const oldRefreshToken = req.cookies.get('admin_session')?.value
  if (!oldRefreshToken) return NextResponse.json({ error: 'No session' }, { status: 401 })

  const existing = await getSession(oldRefreshToken)
  if (!existing) return NextResponse.json({ error: 'Session expired' }, { status: 401 })

  const newRefreshToken = await rotateSession(oldRefreshToken)
  if (!newRefreshToken) return NextResponse.json({ error: 'Session expired' }, { status: 401 })

  const accessToken = await signAccessToken({
    sub: existing.adminId, email: existing.email, role: existing.role, sessionId: newRefreshToken,
  })

  const res = NextResponse.json({ accessToken })
  res.cookies.set('admin_session', newRefreshToken, {
    httpOnly: true, secure: true, sameSite: 'strict', path: '/',
    maxAge: Number(process.env.ADMIN_REFRESH_TOKEN_TTL ?? 28800),
  })
  return res
}
