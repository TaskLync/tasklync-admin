import { NextRequest, NextResponse } from 'next/server'
import { jwtVerify } from 'jose'
import { permissionForPath, hasPermission } from '@/lib/auth/rbac'
import type { AdminRole } from '@/types/auth'

const secret = new TextEncoder().encode(process.env.ADMIN_JWT_SECRET!)

export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl
  const isAuthRoute = pathname.startsWith('/auth')
  const isAuthApi = pathname.startsWith('/api/admin/auth')

  if (isAuthRoute || isAuthApi) return NextResponse.next()

  const accessToken = request.headers.get('authorization')?.replace('Bearer ', '')
    ?? request.cookies.get('admin_access')?.value

  if (!accessToken) {
    return NextResponse.redirect(new URL('/auth/login', request.url))
  }

  try {
    const { payload } = await jwtVerify(accessToken, secret)
    const role = payload.role as AdminRole

    const requiredPermission = permissionForPath(pathname)
    if (requiredPermission && !hasPermission(role, requiredPermission)) {
      return NextResponse.redirect(new URL('/', request.url)) // bounce to dashboard, not an error page
    }

    return NextResponse.next()
  } catch {
    // Invalid, tampered, or expired token â€” never let the request through.
    // Client-side useAdminSession handles silent refresh for API calls;
    // a plain page navigation with a dead token must go straight to login.
    return NextResponse.redirect(new URL('/auth/login', request.url))
  }
}

export const runtime = 'nodejs'

import { NextRequest, NextResponse } from 'next/server'
import { verifyAccessToken } from '@/lib/auth/jwt'
import { db } from '@/lib/db'

export async function GET(req: NextRequest) {
  const auth = req.headers.get('authorization')?.replace('Bearer ', '')
  if (!auth) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const payload = await verifyAccessToken(auth).catch(() => null)
  if (!payload) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const { rows } = await db.query(
    'SELECT id, email, full_name, role, mfa_enabled FROM admin_users WHERE id = $1',
    [payload.sub]
  )
  if (!rows[0]) return NextResponse.json({ error: 'Not found' }, { status: 404 })

  const a = rows[0]
  return NextResponse.json({
    id: a.id, email: a.email, fullName: a.full_name, role: a.role, mfaEnabled: a.mfa_enabled,
  })
}

// lib/rbac-guard.ts
import { NextResponse } from 'next/server'
import { cookies } from 'next/headers'
import { verifyAccessToken } from '@/lib/auth/jwt'
import { hasPermission } from '@/lib/auth/rbac'
import type { AdminRole } from '@/types/auth'

interface AdminContext {
  adminId: string
  adminEmail: string
  role: AdminRole
  sessionId: string
}

type PermissionResult =
  | { ok: true; admin: AdminContext }
  | { ok: false; response: NextResponse }

export async function requirePermission(permission: string): Promise<PermissionResult> {
  const cookieStore = await cookies()
  const token = cookieStore.get('admin_access')?.value

  if (!token) {
    return { ok: false, response: NextResponse.json({ error: 'Unauthorized' }, { status: 401 }) }
  }

  let payload
  try {
    payload = await verifyAccessToken(token)
  } catch {
    return { ok: false, response: NextResponse.json({ error: 'Unauthorized' }, { status: 401 }) }
  }

  if (!hasPermission(payload.role, permission)) {
    return { ok: false, response: NextResponse.json({ error: 'Forbidden' }, { status: 403 }) }
  }

  return {
    ok: true,
    admin: {
      adminId: payload.sub,
      adminEmail: payload.email,
      role: payload.role,
      sessionId: payload.sessionId,
    },
  }
}

==================== lib\auth\session.ts ====================
import { redis } from '@/lib/redis'
import crypto from 'crypto'
import type { AdminRole, SessionRecord } from '@/types/auth'

const REFRESH_TTL = Number(process.env.ADMIN_REFRESH_TOKEN_TTL ?? 28800)
const MAX_SESSIONS = Number(process.env.ADMIN_MAX_CONCURRENT_SESSIONS ?? 2)

function sessionKey(refreshToken: string) {
  return `admin:session:${refreshToken}`
}
function sessionSetKey(adminId: string) {
  return `admin:sessions:${adminId}`
}

export async function createSession(
  adminId: string,
  email: string,
  role: AdminRole,
  ip: string
): Promise<string> {
  const refreshToken = crypto.randomBytes(32).toString('hex')
  const record: SessionRecord = { adminId, email, role, createdAt: new Date().toISOString(), ip }

  await redis.set(sessionKey(refreshToken), JSON.stringify(record), 'EX', REFRESH_TTL)

  // Track sessions per admin, evict oldest if over the concurrent-session limit
  const setKey = sessionSetKey(adminId)
  await redis.zadd(setKey, Date.now(), refreshToken)
  const count = await redis.zcard(setKey)
  if (count > MAX_SESSIONS) {
    const toEvict = await redis.zrange(setKey, 0, count - MAX_SESSIONS - 1)
    if (toEvict.length) {
      await redis.del(...toEvict.map(sessionKey))
      await redis.zrem(setKey, ...toEvict)
    }
  }
  await redis.expire(setKey, REFRESH_TTL)

  return refreshToken
}

export async function getSession(refreshToken: string): Promise<SessionRecord | null> {
  const raw = await redis.get(sessionKey(refreshToken))
  return raw ? JSON.parse(raw) : null
}

export async function rotateSession(oldRefreshToken: string): Promise<string | null> {
  const record = await getSession(oldRefreshToken)
  if (!record) return null

  await redis.del(sessionKey(oldRefreshToken))
  await redis.zrem(sessionSetKey(record.adminId), oldRefreshToken)

  return createSession(record.adminId, record.email, record.role, record.ip)
}

export async function destroySession(refreshToken: string): Promise<void> {
  const record = await getSession(refreshToken)
  await redis.del(sessionKey(refreshToken))
  if (record) await redis.zrem(sessionSetKey(record.adminId), refreshToken)
}

// Maps URL path prefixes -> required permission, used by middleware.ts
// NOTE: order matters. permissionForPath uses find() + startsWith, so more specific prefixes must come first.
export const ROUTE_PERMISSIONS: { prefix: string; permission: string }[] = [
  { prefix: '/workers/verification', permission: 'workers:verify' }, // must stay above '/workers'
  { prefix: '/workers', permission: 'workers:view' },
  { prefix: '/users', permission: 'users:view' },
  { prefix: '/bookings', permission: 'bookings:view' },
  { prefix: '/payments', permission: 'payments:view' },
  { prefix: '/disputes', permission: 'disputes:view' },
  { prefix: '/notifications', permission: 'notifications:view' },
  { prefix: '/analytics', permission: 'analytics:view' },
  { prefix: '/categories', permission: 'categories:view' },
  { prefix: '/settings', permission: 'settings:view' },
]

export function hasPermission(role: AdminRole, permission: string): boolean {
  return PERMISSIONS[permission]?.includes(role) ?? false
}

export function permissionForPath(pathname: string): string | null {
  const match = ROUTE_PERMISSIONS.find((r) => pathname.startsWith(r.prefix))
  return match?.permission ?? null
}

import { useEffect, useState, useCallback } from 'react'
import { useRouter } from 'next/navigation'
import type { AdminUser } from '@/types/auth'

export function useAdminSession() {
  const router = useRouter()
  const [admin, setAdmin] = useState<AdminUser | null>(null)
  const [loading, setLoading] = useState(true)

  const fetchWithAuth = useCallback(async (input: RequestInfo, init: RequestInit = {}) => {
    let token = sessionStorage.getItem('admin_access_token')
    let res = await fetch(input, { ...init, headers: { ...init.headers, Authorization: `Bearer ${token}` } })

    if (res.status === 401) {
      const refreshRes = await fetch('/api/admin/auth/refresh', { method: 'POST' })
      if (!refreshRes.ok) {
        router.push('/auth/login')
        throw new Error('Session expired')
      }
      const { accessToken } = await refreshRes.json()
      sessionStorage.setItem('admin_access_token', accessToken)
      res = await fetch(input, { ...init, headers: { ...init.headers, Authorization: `Bearer ${accessToken}` } })
    }
    return res
  }, [router])

  useEffect(() => {
    fetchWithAuth('/api/admin/auth/session')
      .then((r) => r.json())
      .then(setAdmin)
      .catch(() => setAdmin(null))
      .finally(() => setLoading(false))
  }, [fetchWithAuth])

  return { admin, loading, fetchWithAuth }
}

import ComposeTab, { type ComposeSeed } from './ComposeTab'

const TAB_LABEL: Record<NotificationTab, string> = {
  compose: 'Compose',
  history: 'History',
  templates: 'Templates',
}

const TAB_HINT: Record<NotificationTab, string> = {
  compose: 'Send a push or SMS to a group of users or workers. Booking, payment and chat notifications are automatic and not listed here.',
  history: 'Every broadcast that was sent, with status and recipients. Click one for delivery details.',
  templates: 'Ready-made messages in English and Roman Urdu. Pick one to prefill the composer.',
}

export default function NotificationsSection() {
  const searchParams = useSearchParams()
  const tabParam = searchParams.get('tab')
  const validTab = (NOTIFICATION_TABS as readonly string[]).includes(tabParam ?? '')

  const [tab, setTab] = useState<NotificationTab>(validTab ? (tabParam as NotificationTab) : 'compose')
  const [seed, setSeed] = useState<ComposeSeed | null>(null)
  // bump to remount the composer when a template is chosen from the Templates tab
  const [composeKey, setComposeKey] = useState(0)

  function useTemplate(next: ComposeSeed) {
    setSeed(next)
    setComposeKey((k) => k + 1)
    setTab('compose')
  }

  return (
    <div className="flex flex-col gap-4">
      <div>
        <div className="flex flex-wrap gap-1 border-b border-gray-200">
          {NOTIFICATION_TABS.map((t) => (
            <button
              key={t}
              onClick={() => setTab(t)}
              className={cn(
                'border-b-2 px-3 py-2 text-sm transition-colors duration-150',
                tab === t
                  ? 'border-[#1F6F5F] font-medium text-[#1F6F5F]'
                  : 'border-transparent text-gray-500 hover:text-gray-700'
              )}
            >
              {TAB_LABEL[t]}
            </button>
          ))}
        </div>
        <p className="mt-2 text-xs text-gray-500">{TAB_HINT[tab]}</p>
      </div>

      {tab === 'compose' && <ComposeTab key={composeKey} seed={seed} />}
      {tab === 'history' && <HistoryTab />}
      {tab === 'templates' && <TemplatesTab onUse={useTemplate} />}
    </div>
  )

import { useState } from 'react'
import { toast } from 'sonner'
import { Button } from '@/components/ui/Button'
import { cn } from '@/lib/cn'
import { useWorkerFilterOptions } from '@/hooks/useWorkers'
import { notificationActions } from '@/hooks/useNotifications'
import { LIMITS } from '@/lib/notification-constants'
import type { AudienceSelection, ResolveResult } from '@/types/notification'
import { Field, KNOWN_CITIES, SELECT_CLASS, Segmented } from './notification-ui'

type Mode = 'all_users' | 'all_workers' | 'category_city' | 'custom_ids'

const OPTIONS: { mode: Mode; title: string; hint: string; broad?: boolean }[] = [
  { mode: 'all_users', title: 'All users', hint: 'Every customer account.', broad: true },
  { mode: 'all_workers', title: 'All workers', hint: 'Every worker account.', broad: true },
  { mode: 'category_city', title: 'Workers by service & city', hint: 'Verified workers in one category and city.' },
  { mode: 'custom_ids', title: 'Specific people', hint: 'Paste phone numbers, emails or IDs. Good for tests.' },
]

interface Props {
  value: AudienceSelection | null
  onChange: (value: AudienceSelection | null) => void
  /** SUPER_ADMIN only. */
  canBroadcastAll: boolean
}

export default function AudiencePicker({ value, onChange, canBroadcastAll }: Props) {
  const { categories } = useWorkerFilterOptions()
  const [mode, setMode] = useState<Mode | null>(value?.segmentType ?? null)
  const [categoryId, setCategoryId] = useState('')
  const [city, setCity] = useState('')
  const [who, setWho] = useState<'users' | 'workers'>('users')
  const [text, setText] = useState('')
  const [lookup, setLookup] = useState<ResolveResult | null>(null)
  const [looking, setLooking] = useState(false)

  function pick(m: Mode) {
    setMode(m)
    setLookup(null)
    if (m === 'all_users' || m === 'all_workers') onChange({ segmentType: m })
    else if (m === 'category_city')
      onChange(categoryId && city ? { segmentType: 'category_city', categoryId, city } : null)
    else onChange(null)
  }

  function setCategoryCity(nextCategory: string, nextCity: string) {
    setCategoryId(nextCategory)
    setCity(nextCity)
    onChange(
      nextCategory && nextCity ? { segmentType: 'category_city', categoryId: nextCategory, city: nextCity } : null
    )
  }

  function identifiers(): string[] {
    return [...new Set(text.split(/[\n,;]+/).map((s) => s.trim()).filter(Boolean))]
  }

  function invalidateLookup() {
    setLookup(null)
    onChange(null)
  }

  async function runLookup() {
    const list = identifiers()
    if (list.length === 0) return toast.error('Paste at least one phone, email or ID')
    if (list.length > LIMITS.specificPeople) {
      return toast.error(`At most ${LIMITS.specificPeople} people at a time`)
    }
    setLooking(true)
    try {
      const res = await notificationActions.resolveRecipients(who, list)
      setLookup(res)
      onChange(
        res.matched.length
          ? { segmentType: 'custom_ids', audience: who, ids: res.matched.map((m) => m.id) }
          : null
      )
    } catch (e) {
      toast.error((e as Error).message)
    } finally {
      setLooking(false)
    }
  }

  return (
    <div className="flex flex-col gap-4">
      <div className="grid grid-cols-1 gap-2 sm:grid-cols-2">
        {OPTIONS.map((o) => {
          const disabled = Boolean(o.broad) && !canBroadcastAll
          return (
            <label
              key={o.mode}
              className={cn(
                'flex items-start gap-2 rounded-md border p-3 text-sm',
                disabled ? 'cursor-not-allowed border-gray-100 bg-gray-50 opacity-60' : 'cursor-pointer',
                !disabled && mode === o.mode ? 'border-[#1F6F5F] bg-[#1F6F5F]/5' : 'border-gray-200'
              )}
            >
              <input
                type="radio"
                name="audience"
                className="mt-0.5 accent-[#1F6F5F]"
                disabled={disabled}
                checked={mode === o.mode}
                onChange={() => pick(o.mode)}
              />
              <span>
                <span className="font-medium text-gray-900">{o.title}</span>
                <span className="block text-xs text-gray-400">
                  {disabled ? 'Super Admin only.' : o.hint}
                </span>
              </span>
            </label>
          )
        })}
      </div>

      {mode === 'category_city' && (
        <div className="flex flex-wrap gap-3">
          <Field label="Service">
            <select
              className={SELECT_CLASS}
              value={categoryId}
              onChange={(e) => setCategoryCity(e.target.value, city)}
            >
              <option value="">Choose a service</option>
              {categories.map((c) => (
                <option key={c.id} value={c.id}>
                  {c.name}
                </option>
              ))}
            </select>
          </Field>
          <Field label="City">
            <select
              className={SELECT_CLASS}
              value={city}
              onChange={(e) => setCategoryCity(categoryId, e.target.value)}
            >
              <option value="">Choose a city</option>
              {KNOWN_CITIES.map((c) => (
                <option key={c} value={c}>
                  {c}
                </option>
              ))}
            </select>
          </Field>
        </div>
      )}

      {mode === 'custom_ids' && (
        <div className="flex flex-col gap-3">
          <Segmented
            value={who}
            options={[
              { value: 'users', label: 'Users' },
              { value: 'workers', label: 'Workers' },
            ]}
            onChange={(v) => {
              setWho(v)
              invalidateLookup()
            }}
          />
          <textarea
            className="h-28 w-full rounded-md border border-gray-300 px-3 py-2 font-mono text-xs focus:outline-none focus:ring-2 focus:ring-[#1F6F5F]"
            placeholder={'0300 1234567\n+92 321 7654321\nname@example.com'}
            value={text}
            onChange={(e) => {
              setText(e.target.value)
              if (lookup) invalidateLookup()
            }}
          />
          <div className="flex items-center gap-3">
            <Button variant="secondary" size="sm" loading={looking} onClick={runLookup}>
              Look up people
            </Button>
            <span className="text-xs text-gray-400">
              {identifiers().length} / {LIMITS.specificPeople}. One per line or comma-separated.
            </span>
          </div>

          {lookup && (
            <div className="rounded-md bg-gray-50 px-3 py-2 text-sm">
              <p className="font-medium text-gray-800">
                {lookup.matched.length} found
                {lookup.notFound.length > 0 && (
                  <span className="font-normal text-amber-700"> Â· {lookup.notFound.length} not found</span>
                )}
              </p>
              {lookup.matched.length > 0 && (
                <ul className="mt-1 max-h-32 overflow-y-auto text-xs text-gray-600">
                  {lookup.matched.slice(0, 20).map((m) => (
                    <li key={m.id}>
                      {m.name ?? 'Unnamed'} <span className="font-mono text-gray-400">{m.phone ?? m.email ?? ''}</span>
                    </li>

                </ul>
              )}
              {lookup.notFound.length > 0 && (
